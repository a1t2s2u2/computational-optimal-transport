\subsection*{空間}
\begin{tabular}{ll}
  $X$ & $\R^n$ のコンパクト凸部分集合 \\
  $\Borel(X)$ & $X$ のボレル $\sigma$-代数 \\
  $\Leb^n$ & $\R^n$ 上のルベーグ測度 \\
  $\Haus^k$ & $\R^n$ 上の $k$ 次元ハウスドルフ測度 \\
  $\Lip(X)$ & $X$ 上の実数値リプシッツ関数全体 \\
  $\Lip_1(X)$ & リプシッツ定数が $1$ 以下のリプシッツ関数全体 \\
\end{tabular}

\subsection*{測度}
\begin{tabular}{ll}
  $\Meas(X)$ & 有限全変動をもつ $X$ 上の符号付きラドン測度全体 \\
  $\Meas_+(X)$ & 正の有限ラドン測度全体 \\
  $\Prob(X)$ & $X$ 上の確率測度全体 \\
  $\abs{\mu}$ & $\mu \in [\Meas(X)]^n$ の全変動 \\
  $\mu^+,\; \mu^-$ & $\mu \in \Meas(X)$ の正部分・負部分 \\
  $f\pushforward \mu$ & ボレル写像 $f$ による $\mu$ の押し出し \\
\end{tabular}

\subsection*{最適輸送}
\begin{tabular}{ll}
  $\mu,\; \nu \in \Prob(X)$ & 輸送元・輸送先の確率測度 \\
  $\psi \colon X \to X$ & 輸送写像（$\psi\pushforward \mu = \nu$） \\
  $\Gamma(\mu, \nu)$ & 輸送計画（カップリング）の集合 \\
  $\gamma \in \Prob(X \times X)$ & 輸送計画（周辺条件 $\pi_{0\#}\gamma = \mu$, $\pi_{1\#}\gamma = \nu$）\\
  $c \colon X \times X \to [0,\infty]$ & コスト関数 \\
  $I(\gamma)$ & 輸送コスト汎関数 $\int_{X \times X} c(x,y)\,\d\gamma(x,y)$ \\
  $\Fcost_c(\mu, \nu)$ & 最適輸送コスト $\min_\gamma I(\gamma)$ \\
  $\Fcost_p(\mu, \nu)$ & $p$-ワッサーシュタイン距離 \\
\end{tabular}
