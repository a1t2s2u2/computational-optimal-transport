\chapter{最適輸送計画：存在と正則性}
\label{ch:existence-regularity}

\begin{remark}{記法と用語の読み替え}{amb-notation-conv}
  本ノートは Ambrosio の講義ノートに沿うが，以下の読み替えを行う．
  \begin{itemize}
    \item 確率測度を $f_0, f_1$ でなく $\mu, \nu$ と書く．
    \item 原文の ``Kantorovich–Wasserstein distance'' は単に Wasserstein 距離と呼ぶ．
    \item Wasserstein 距離の記号を $\Fcost_p$ でなく $\Wass_p$ と書く．
  \end{itemize}
\end{remark}

$X$ を $\R^n$ のコンパクト凸部分集合とし，$\mu, \nu \in \Prob(X)$ とする．

\section{Kantorovich 問題}
\label{sec:kantorovich}

\begin{definition}{Kantorovich 問題 (MK)}{amb-mk}
  \[
    \Gamma(\mu,\nu) := \left\{\,
      \gamma \in \Prob(X \times X)
      \;\middle|\;
      \begin{aligned}
        &\gamma(A \times X) = \mu(A) \quad \forall\, A \in \Borel(X) \\
        &\gamma(X \times B) = \nu(B) \quad \forall\, B \in \Borel(X)
      \end{aligned}
    \,\right\}
  \]
  とおく．Borel コスト関数 $c \colon X \times X \to [0, \infty]$ に対し，
  \[
    \Fcost_c(\mu, \nu) :=
    \min_{\gamma \in \Gamma(\mu,\nu)}
    \int_{X \times X} c(x,y)\,\d\gamma(x,y)
  \]
  を \textbf{Kantorovich 問題} (MK) と呼ぶ．
\end{definition}

Kantorovich の設定では $\mu, \nu$ に制約はなく，直積 $\mu \otimes \nu$ が常に
$\Gamma(\mu,\nu)$ に属するから，$\Gamma(\mu,\nu) \neq \emptyset$ である．
これにより次のような距離の族が定まる．

\begin{definition}{Wasserstein 距離}{amb-wasserstein}
  $p \geq 1$ とし，$\mu, \nu \in \Prob(X)$ とする．
  $\mu$ と $\nu$ の間の \textbf{$p$-Wasserstein 距離}を
  \[
    \Wass_p(\mu, \nu)
    := \left(
      \min_{\gamma \in \Gamma(\mu,\nu)}
      \int_{X \times X} \abs{x - y}^p \,\d\gamma
    \right)^{1/p}
  \]
  で定める．
\end{definition}

\section{最適計画の存在}
\label{sec:existence}

最適計画の存在は，$\Gamma(\mu,\nu)$ の $w^*$-コンパクト性と
$I$ の下半連続性から従う．

\begin{theorem}{最適計画の存在}{amb-existence}
  \blockmeta{uses=def:amb-mk}
  $c$ が $X \times X$ 上で下半連続であると仮定する．
  このとき (MK) を達成する $\gamma \in \Gamma(\mu,\nu)$ が存在する．
\end{theorem}

\begin{proof}
  $\Gamma(\mu,\nu)$ は $w^*$-位相で閉かつ有界であり，$X \times X$ がコンパクトだから $w^*$-コンパクトである．

  $c$ が下半連続のとき，連続関数の増大列 $c_h \uparrow c$ で
  下から近似でき，$c_h$ の誘導する汎関数 $I_h$ は
  $w^*$-連続かつ $I_h \uparrow I$ だから，$I$ は下半連続である．
  コンパクト集合上の下半連続関数は最小値を達成する．
\end{proof}

\section{最適計画の正則性と cyclical monotonicity}
\label{sec:regularity}

最適計画の台（サポート）には強い構造が備わる．
それを記述するために，$c$-cyclical monotonicity を導入する．

\begin{definition}{$c$-cyclical monotonicity}{amb-cyclic-monotonicity}
  $\Gamma \subset X \times X$ が \textbf{$c$-cyclically monotone} であるとは，
  任意の $n \geq 2$ と任意の $(x_i, y_i) \in \Gamma$（$1 \leq i \leq n$）に対し
  \[
    \sum_{i=1}^{n} c(x_{i+1}, y_i) \geq \sum_{i=1}^{n} c(x_i, y_i)
  \]
  が成り立つことをいう．ただし $x_{n+1} = x_1$ とする．
\end{definition}

\begin{remark}{cyclical monotonicity の置換表示}{amb-cyclic-perm}
  cyclical monotonicity は次の同値な形で述べられる：
  任意の置換 $\sigma \colon \{1,\ldots,n\} \to \{1,\ldots,n\}$ に対し
  \[
    \sum_{i=1}^{n} c(x_{\sigma(i)}, y_i) \geq \sum_{i=1}^{n} c(x_i, y_i).
  \]
  これは，添え字の巡回置換だけでなく一般の置換でもコストが悪化しない
  ことを意味する．
\end{remark}

\begin{theorem}{最適計画の正則性}{amb-regularity}
  \blockmeta{uses=def:amb-mk; def:amb-cyclic-monotonicity}
  $c$ が連続かつ実数値であると仮定する．
  任意の最適 $\gamma$ に対し $\spt\,\gamma$ は $c$-cyclically monotone である．

  さらに，全ての最適計画の $\spt$ の和集合の閉包も $c$-cyclically monotone である．
\end{theorem}

\begin{proof}
  最適な $\gamma$ の台が $c$-cyclically monotone でないと仮定し，矛盾を導く．
  $(x_i, y_i) \in \spt\,\gamma$（$i = 1, \ldots, n$）で
  \[
    f\bigl((x_i), (y_i)\bigr) := \sum_{i=1}^n c(x_{i+1}, y_i) - c(x_i, y_i) < 0
  \]
  を満たすものが存在するとする．$c$ の連続性から，各 $(x_i, y_i)$ の
  近傍 $U_i \times V_i$ 上でも $f < 0$ が保たれ，
  $\gamma(U_i \times V_i) > 0$ である．

  $\lambda = \min_i \gamma(U_i \times V_i)$ とし，
  $\gamma_i \in \Prob(U_i \times V_i)$ を $\gamma$ の $U_i \times V_i$ への
  正規化された制限とする．
  コンパクト空間 $Y$ と確率測度 $\sigma$ を使い，
  各 $\gamma_i = \eta_{i\#}\sigma$（$\eta_i = u_i \times v_i$）と表して
  \[
    \gamma' := \gamma + \frac{\lambda}{n} \sum_{i=1}^n
    \bigl[(u_{i+1} \times v_i)\pushforward \sigma - (u_i \times v_i)\pushforward \sigma\bigr]
  \]
  とおけば $\gamma' \in \Meas_+(X \times X)$ かつ周辺条件は保たれ，
  \[
    I(\gamma') - I(\gamma) = \frac{\lambda}{n}
    \int_Y \sum_{i=1}^n \bigl[c(u_{i+1}, v_i) - c(u_i, v_i)\bigr]\,\d\sigma < 0
  \]
  となり $\gamma$ の最適性に矛盾する．

  後半は，最適計画の集合が $w^*$-コンパクトであることと，
  稠密な列 $(\gamma_h)$ の $\spt$ の合併
  $\bigcup_{h=1}^N \spt\,\gamma_h = \spt\bigl(\sum_{h=1}^N \frac{1}{N}\gamma_h\bigr)$
  が（凸結合も最適だから）$c$-cyclically monotone であることから，
  極限操作で閉包まで拡張される．
\end{proof}
