\chapter{Wasserstein 距離 \texorpdfstring{$W_p$}{Wp}}
\label{ch:wasserstein-metrics}

\section{設定と定義}
\label{sec:wp-definition}

\begin{definition}{確率測度の空間}{gs-probability-space}
  $(X,d)$ を完備かつ可分な距離空間とする．このような空間を
  \textbf{Polish 空間}という．
  $X$ の開集合から生成される $\sigma$-代数を $\Borel(X)$ と書き，
  \[
    \Prob(X)
    :=
    \left\{
      \mu\mid \mu\text{ は }\Borel(X)\text{ 上の測度で }\mu(X)=1
    \right\}
  \]
  と定める．すなわち $\Prob(X)$ は $X$ 上の Borel 確率測度全体である．
\end{definition}

\begin{definition}{有限 $p$ 次モーメントをもつ確率測度}{gs-p-space}
  任意に点 $x_0\in X$ を一つ固定する．$1\leq p<\infty$ に対し
  \[
    \Pp{p}(X)
    :=
    \left\{
      \mu\in\Prob(X)
      \;\middle|\;
      \int_X d(x,x_0)^p\,\d\mu(x)<\infty
    \right\}
  \]
  と定める．また
  \[
    \Pp{\infty}(X)
    :=
    \left\{\mu\in\Prob(X)\mid \supp\mu\text{ は有界}\right\}
  \]
  とおく．ただし，$\mu$ の\textbf{台}とは
  \[
    \supp\mu
    =
    \left\{x\in X\mid x\text{ の任意の開近傍 }U\text{ に対して }\mu(U)>0\right\}
  \]
  のことである．
  $d(x,x_0')\leq d(x,x_0)+d(x_0,x_0')$ より，
  $\Pp{p}(X)$ は点 $x_0$ の選び方によらない．
\end{definition}

\begin{definition}{Coupling と Wasserstein 距離}{gs-wasserstein}
  $\mu,\nu\in\Prob(X)$ の\textbf{coupling}の集合を
  \[
    \Couplings(\mu,\nu)
    :=
    \left\{
      \pi\in\Prob(X\times X)
      \;\middle|\;
      \begin{aligned}
        \pi(A\times X)&=\mu(A) &&(\forall A\in\Borel(X)),\\
        \pi(X\times B)&=\nu(B) &&(\forall B\in\Borel(X))
      \end{aligned}
    \right\}
  \]
  とする．$1\leq p<\infty$ と $\mu,\nu\in\Pp{p}(X)$ に対し
  \[
    \Wass_p(\mu,\nu)
    :=
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \left(
      \int_{X\times X}d(x,y)^p\,\d\pi(x,y)
    \right)^{1/p},
  \]
  また $\mu,\nu\in\Pp{\infty}(X)$ に対し
  \[
    \Wass_\infty(\mu,\nu)
    :=
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \esssup_{(x,y)\sim\pi}d(x,y)
  \]
  と定める．ここで本質的上限は，
  $\pi(\{(x,y)\mid d(x,y)\leq R\})=1$ となる最小の $R$ である．
\end{definition}

$\Pp{p}(X)$ への制限は，$\Wass_p$ を有限値にするために必要である．
実際，点 $x_0$ に質量 $1$ を置く確率測度を $\delta_{x_0}$ と書けば
\[
  \Wass_p(\mu,\delta_{x_0})^p
  =
  \int_X d(x,x_0)^p\,\d\mu(x)
\]
となる．

$p<\infty$ のとき，直積 coupling $\mu\otimes\nu$ と
\[
  d(x,y)^p
  \leq
  2^{p-1}\bigl(d(x,x_0)^p+d(y,x_0)^p\bigr)
\]
を用いれば，$\mu,\nu\in\Pp{p}(X)$ に対して $\Wass_p(\mu,\nu)<\infty$ が分かる．

\section{最適 coupling の存在}
\label{sec:optimal-coupling}

確率測度の\textbf{弱収束} $\pi_n\rightharpoonup\pi$ とは，
すべての有界連続関数 $f$ に対して
$\int f\,\d\pi_n\to\int f\,\d\pi$ となることをいう．

\begin{lemma}{輸送コストの下半連続性}{gs-cost-lsc}
  $\pi_n\rightharpoonup\pi$ ならば，$1\leq p<\infty$ に対し
  \[
    \int d(x,y)^p\,\d\pi(x,y)
    \leq
    \liminf_{n\to\infty}
    \int d(x,y)^p\,\d\pi_n(x,y).
  \]
\end{lemma}

\begin{proof}
  $(x,y)\mapsto d(x,y)^p$ は非負かつ連続なので，
  Portmanteau の定理から従う．
\end{proof}

\begin{proposition}{最適 coupling の存在（原論文 Proposition 1）}{gs-optimal-coupling}
  $1\leq p\leq\infty$ とする．
  $\mu,\nu$ が $\Wass_p$ の定義域に属するならば，
  下限を達成する $\pi^\star\in\Couplings(\mu,\nu)$ が存在する．
\end{proposition}

\begin{proof}
  以下の3点を示す：
  (i) $\Couplings(\mu,\nu)\neq\emptyset$，
  (ii) $\Couplings(\mu,\nu)$ は tight，
  (iii) 弱極限が下限を達成する．

  \textbf{(i) 非空性．}
  直積測度 $\mu\otimes\nu$ は
  $(\mu\otimes\nu)(A\times X)=\mu(A)$，
  $(\mu\otimes\nu)(X\times B)=\nu(B)$
  を満たすから $\mu\otimes\nu\in\Couplings(\mu,\nu)$．

  \textbf{(ii) Tightness．}
  Polish 空間上の Borel 確率測度は tight である（Ulam の定理）．
  任意の $\varepsilon>0$ に対し，
  コンパクト集合 $K_1,K_2\subset X$ で
  $\mu(K_1)>1-\varepsilon$，$\nu(K_2)>1-\varepsilon$ なるものを取る．
  任意の $\pi\in\Couplings(\mu,\nu)$ に対し，
  \[
    \pi(K_1\times K_2)
    = 1 - \pi(K_1^c\times X \cup X\times K_2^c)
    \geq 1 - \pi(K_1^c\times X) - \pi(X\times K_2^c)
    = 1 - \mu(K_1^c) - \nu(K_2^c)
    > 1 - 2\varepsilon.
  \]
  $K_1\times K_2$ は $X\times X$ のコンパクト集合だから，
  $\Couplings(\mu,\nu)$ は tight である．

  \textbf{(iii) 下限の達成．}
  $p<\infty$ とする．最小化列
  $(\pi_n)\subset\Couplings(\mu,\nu)$ を取る．
  (ii) と Prokhorov の定理により，部分列を取れば
  $\pi_{n_k}\rightharpoonup\pi^\star$ とできる．
  弱極限の周辺分布は保たれるから
  $\pi^\star\in\Couplings(\mu,\nu)$ であり，
  補題~\ref{lem:gs-cost-lsc}より $\pi^\star$ は下限を達成する．

  $p=\infty$ では
  \[
    \esssup_{(x,y)\sim\pi_n}d(x,y)
    \leq
    \Wass_\infty(\mu,\nu)+\frac1n
  \]
  を満たす coupling を取り，弱収束部分列 $\pi_{n_k}\rightharpoonup\pi^\star$ を取る．
  各 $m$ に対し，十分大きな $k$ で
  $\pi_{n_k}(\{(x,y)\mid d(x,y)\leq\Wass_\infty(\mu,\nu)+1/m\})=1$ である．
  この集合は閉なので，Portmanteau の定理から
  $\pi^\star(\{(x,y)\mid d(x,y)\leq\Wass_\infty(\mu,\nu)+1/m\})=1$．
  $m\to\infty$ とすれば $\pi^\star$ は下限を達成する．
\end{proof}

\section{Wasserstein 距離の距離性}
\label{sec:metric-property}

\begin{lemma}{Coupling の貼り合わせ（Gluing lemma）}{gs-gluing}
  $\mu,\nu,\rho\in\Prob(X)$ とし，
  $\pi\in\Couplings(\mu,\nu)$，$\sigma\in\Couplings(\nu,\rho)$ とする．
  このとき $X^3$ 上の確率測度 $\boldsymbol{\gamma}$ で，
  次の2条件を同時に満たすものが存在する：
  \begin{enumerate}
    \item $(x,y)$ への周辺分布が $\pi$，
    \item $(y,z)$ への周辺分布が $\sigma$．
  \end{enumerate}
\end{lemma}

\begin{proof}
  正則条件付き確率を用いて $\pi$ と $\sigma$ を
  共通の周辺 $\nu$ で分解する：
  \[
    \pi(\d x,\d y)
    =K_{\pi}(y,\d x)\,\nu(\d y),
    \qquad
    \sigma(\d y,\d z)
    =\nu(\d y)\,K_{\sigma}(y,\d z).
  \]
  ここで $K_\pi(y,\cdot)$，$K_\sigma(y,\cdot)$ は
  それぞれ $\pi$，$\sigma$ の $y$ を固定したときの条件付き分布である．
  \[
    \boldsymbol{\gamma}(\d x,\d y,\d z)
    :=
    K_{\pi}(y,\d x)\,\nu(\d y)\,K_{\sigma}(y,\d z)
  \]
  とおけば，$(x,y)$ 周辺は $\pi$，$(y,z)$ 周辺は $\sigma$ となる．
\end{proof}

\begin{proposition}{Wasserstein 距離は距離である（原論文 Proposition 2）}{gs-wp-metric}
  $1\leq p\leq\infty$ に対し，
  $\Wass_p$ は $\Pp{p}(X)$ 上の距離である．
\end{proposition}

\begin{proof}
  距離の公理 (1) 非負性，(2) 対称性，(3) 非退化性，(4) 三角不等式を順に示す．
  $p<\infty$ の場合を述べ，$p=\infty$ は各段階で本質的上限に置き換えればよい．

  \textbf{(1) 非負性．}
  $d\geq 0$ より $\int d^p\,\d\pi\geq 0$ だから $\Wass_p\geq 0$．

  \textbf{(2) 対称性．}
  写像 $s\colon(x,y)\mapsto(y,x)$ について，
  $\pi\in\Couplings(\mu,\nu)$ ならば
  $s_\#\pi\in\Couplings(\nu,\mu)$ であり，
  $d(x,y)=d(y,x)$ より輸送コストは等しい．
  この対応は全単射だから $\Wass_p(\mu,\nu)=\Wass_p(\nu,\mu)$．

  \textbf{(3) 非退化性．}
  $\Wass_p(\mu,\mu)=0$ は対角 coupling
  $(\Id,\Id)_\#\mu$ のコストが $0$ であることから従う．
  逆に $\Wass_p(\mu,\nu)=0$ とする．
  命題~\ref{prop:gs-optimal-coupling}の最適 coupling $\pi^\star$ は
  $\int d(x,y)^p\,\d\pi^\star=0$ を満たす．
  $d^p\geq 0$ より
  $\pi^\star(\{(x,y)\mid d(x,y)>0\})=0$，
  すなわち $\pi^\star$ は対角集合
  $\Delta=\{(x,x)\mid x\in X\}$ に集中する．
  任意の $A\in\Borel(X)$ に対し
  \[
    \mu(A)
    = \pi^\star(A\times X)
    = \pi^\star\bigl((A\times X)\cap\Delta\bigr)
    = \pi^\star\bigl(X\times A)\cap\Delta\bigr)
    = \pi^\star(X\times A)
    = \nu(A)
  \]
  だから $\mu=\nu$．

  \textbf{(4) 三角不等式．}
  $\mu,\nu,\rho\in\Pp{p}(X)$ とし，
  $\pi\in\Couplings(\mu,\nu)$，$\sigma\in\Couplings(\nu,\rho)$
  をそれぞれの最適 coupling とする．
  補題~\ref{lem:gs-gluing}より，
  $X^3$ 上の確率測度 $\boldsymbol{\gamma}$ で
  $(x,y)$ 周辺が $\pi$，$(y,z)$ 周辺が $\sigma$ であるものが存在する．
  $\boldsymbol{\gamma}$ の $(x,z)$ 周辺分布を
  $\tau$ とおくと $\tau\in\Couplings(\mu,\rho)$ だから，
  \[
    \Wass_p(\mu,\rho)
    \leq \norm{d(x,z)}_{L^p(\tau)}
    = \norm{d(x,z)}_{L^p(\boldsymbol{\gamma})}.
  \]
  距離 $d$ の三角不等式 $d(x,z)\leq d(x,y)+d(y,z)$ と
  $L^p$ ノルムの単調性から
  \[
    \norm{d(x,z)}_{L^p(\boldsymbol{\gamma})}
    \leq \norm{d(x,y)+d(y,z)}_{L^p(\boldsymbol{\gamma})}.
  \]
  Minkowski の不等式を適用し，
  周辺分布が $\pi,\sigma$ であることから
  \[
    \norm{d(x,y)+d(y,z)}_{L^p(\boldsymbol{\gamma})}
    \leq \norm{d(x,y)}_{L^p(\pi)}
    + \norm{d(y,z)}_{L^p(\sigma)}
    = \Wass_p(\mu,\nu)+\Wass_p(\nu,\rho).
  \]
\end{proof}
