\chapter{Wasserstein 距離 \texorpdfstring{$W_p$}{Wp}}
\label{ch:wasserstein-metrics}

\section{設定と定義}
\label{sec:wp-definition}

$(X,d)$ を Polish 空間
（完備かつ可分な距離空間：定義~\ref{def:meas-polish}）とする．
Borel $\sigma$-代数 $\Borel(X)$ 上の確率測度全体を
\[
  \Prob(X)
  :=
  \left\{
    \mu\colon\Borel(X)\to[0,1]
    \;\middle|\;
    \mu\text{ は測度},\;\mu(X)=1
  \right\}
\]
とおく．

\begin{definition}{有限 $p$ 次モーメントをもつ確率測度}{wp-p-space}
  任意に点 $x_0\in X$ を一つ固定する．$1\leq p<\infty$ に対し
  \[
    \Pp{p}(X)
    :=
    \left\{
      \mu\in\Prob(X)
      \;\middle|\;
      \int_X d(x,x_0)^p\,\d\mu(x)<\infty
    \right\}
  \]
  と定める．また
  \begin{align*}
    \Pp{\infty}(X)
    &:=
    \left\{\mu\in\Prob(X)\mid \supp\mu\text{ は有界}\right\}\\
    &=
    \left\{
      \mu\in\Prob(X)
      \;\middle|\;
      \sup_{x\in\supp\mu}d(x,x_0)<\infty
    \right\}
  \end{align*}
  とおく（台 $\supp\mu$ は定義~\ref{def:meas-support}，
  有界集合は定義~\ref{def:meas-open-ball}を参照．
  2つの表示の一致は注意~\ref{rem:wp-p-infty-equiv}で確かめる）．
\end{definition}

\begin{lemma}{$\Pp{p}(X)$ の $x_0$ 非依存性}{wp-p-indep}
  $1\leq p<\infty$ のとき，$\Pp{p}(X)$ は点 $x_0$ の選び方によらない．
  （$p=\infty$ の場合，第1の表示（台の有界性）が $x_0$ に言及しないので，
  非依存性は明らかである．）
\end{lemma}

\begin{proof}
  $x_0,x_0'\in X$ と $\mu\in\Prob(X)$ を任意に取り，
  $\int_X d(x,x_0)^p\,\d\mu(x)<\infty$ を仮定する．
  三角不等式から
  \[
    d(x,x_0')
    \leq d(x,x_0)+d(x_0,x_0').
  \]
  両辺 $0$ 以上なので $p$ 乗して
  \[
    d(x,x_0')^p
    \leq
    \bigl(d(x,x_0)+d(x_0,x_0')\bigr)^p.
  \]
  ここで $a:=d(x,x_0)$，$b:=d(x_0,x_0')$ とおけば，
  $f(t)=t^p$（$p\geq 1$）が $[0,\infty)$ 上で凸であること
  （注意~\ref{rem:meas-real-facts}）から
  \begin{align*}
    \text{右辺}
    = (a+b)^p
    &= 2^p\left(\frac{a}{2}+\frac{b}{2}\right)^p\\
    &\leq 2^p\left(\frac{a^p}{2}+\frac{b^p}{2}\right)
    = 2^{p-1}(a^p+b^p).
  \end{align*}
  よって
  \[
    d(x,x_0')^p
    \leq
    2^{p-1}\bigl(d(x,x_0)^p+d(x_0,x_0')^p\bigr).
  \]
  両辺を $\d\mu(x)$ で積分すると
  （右辺第2項は定数なので，$\mu(X)=1$ より積分値は $d(x_0,x_0')^p$）
  \[
    \int_X d(x,x_0')^p\,\d\mu(x)
    \leq
    2^{p-1}\left(\int_X d(x,x_0)^p\,\d\mu(x)+d(x_0,x_0')^p\right)
    <\infty.
  \]
  $x_0$ と $x_0'$ を入れ替えれば逆も成り立つ．
\end{proof}

\begin{remark}{$\Pp{\infty}(X)$ の言い換え}{wp-p-infty-equiv}
  定義~\ref{def:wp-p-space}の $\Pp{\infty}(X)$ の2つの表示が
  一致することを確かめる．
  なお $X$ は可分で $\mu$ は確率測度だから
  $\mu(\supp\mu)=1$（補題~\ref{lem:meas-support-full}），
  特に $\supp\mu\neq\emptyset$ であり，
  上限が $\sup\emptyset=-\infty$ になることはない．
  \begin{itemize}
    \item $\supp\mu$ が有界，すなわちある開球 $B_r(z)$ に含まれるとする．
      三角不等式より $\supp\mu\subset B_{r+d(z,x_0)}(x_0)$ だから，
      $R:=r+d(z,x_0)$ とおけば，すべての $x\in\supp\mu$ で
      $d(x,x_0)<R$，よって
      $\sup_{x\in\supp\mu}d(x,x_0)\leq R<\infty$ である．
    \item 逆に $S:=\sup_{x\in\supp\mu}d(x,x_0)<\infty$ とすると，
      すべての $x\in\supp\mu$ で $d(x,x_0)\leq S<S+1$，
      すなわち $\supp\mu\subset B_{S+1}(x_0)$ だから
      $\supp\mu$ は有界である．
  \end{itemize}
\end{remark}

\begin{definition}{Coupling 集合}{wp-coupling}
  $\mu,\nu\in\Prob(X)$ に対して，\textbf{Coupling 集合}を
  \[
    \Couplings(\mu,\nu)
    :=
    \left\{
      \pi\in\Prob(X\times X)
      \;\middle|\;
      \begin{aligned}
        \pi(A\times X)&=\mu(A) &&(\forall A\in\Borel(X)),\\
        \pi(X\times B)&=\nu(B) &&(\forall B\in\Borel(X))
      \end{aligned}
    \right\}
  \]
  と定める．
  $\pi\in\Couplings(\mu,\nu)$ は $\mu,\nu$ の
  \textbf{coupling}（結合分布）あるいは\textbf{輸送計画}とよばれる．
  この条件は「$\pi$ の第1周辺分布が $\mu$，第2周辺分布が $\nu$」，
  すなわち射影の像測度として
  $\pi\circ\mathrm{pr}_1^{-1}=\mu$，$\pi\circ\mathrm{pr}_2^{-1}=\nu$
  と言い換えられる（補題~\ref{lem:meas-marginal-integral}）．
\end{definition}

\begin{remark}{記法：$(x,y)\sim\pi$ と a.e.}{wp-sampling-notation}
  $\pi\in\Prob(X\times X)$ に対し，$(x,y)\sim\pi$ は
  「確率測度 $\pi$ に従って選ばれた点対 $(x,y)$」を表す文献の記法で，
  $\int_{X\times X} f(x,y)\,\d\pi(x,y)$ は
  $\mathbb{E}_{(x,y)\sim\pi}[f(x,y)]$ とも書かれる
  （本資料では積分記法を用いる）．
  性質が $\pi$-a.e. で成り立つ，の意味は
  定義~\ref{def:meas-ae}の通りである．
\end{remark}

\begin{definition}{Wasserstein 距離}{wp-wasserstein}
  $1\leq p<\infty$ と $\mu,\nu\in\Pp{p}(X)$ に対し，
  下限（定義~\ref{def:meas-sup-inf}）を用いて
  \[
    \Wass_p(\mu,\nu)
    :=
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \left(
      \int_{X\times X}d(x,y)^p\,\d\pi(x,y)
    \right)^{1/p},
  \]
  また $\mu,\nu\in\Pp{\infty}(X)$ に対し，
  $L^\infty$ ノルム（定義~\ref{def:meas-lp-norm}）を用いて
  \[
    \Wass_\infty(\mu,\nu)
    :=
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \norm{d}_{L^\infty(\pi)}
  \]
  と定める．
\end{definition}

\begin{remark}{記法と可測性}{wp-welldef}
  距離関数 $d\colon X\times X\to[0,\infty)$ は，三角不等式から従う評価
  \[
    \abs{d(x,y)-d(x',y')}\leq d(x,x')+d(y,y')
  \]
  により連続，したがって Borel 可測である
  （注意~\ref{rem:meas-measurable-ops}）．
  $d\geq0$ より $1\leq p<\infty$ では
  $\norm{d}_{L^p(\pi)}=\bigl(\int d^p\,\d\pi\bigr)^{1/p}$ だから，
  $1\leq p\leq\infty$ を通じて
  \[
    \Wass_p(\mu,\nu)
    =\inf_{\pi\in\Couplings(\mu,\nu)}\norm{d}_{L^p(\pi)}
  \]
  と統一的に書ける．
  また主張~\ref{clm:meas-esssup-ae}より，$\norm{d}_{L^\infty(\pi)}$ が
  有限のとき（すぐ後の補題~\ref{lem:wp-finite}により
  $\mu,\nu\in\Pp{\infty}(X)$ の coupling では常に有限），それ自身が
  「$d(x,y)\leq R$ が $\pi$-a.e. で成り立つ」をみたす最小の $R\geq0$ であり，
  これは $\esssup_{(x,y)\sim\pi}d(x,y)$ とも書かれる．
  さらに $t\mapsto t^{1/p}$ が $[0,\infty)$ 上の単調な全単射であることから，
  $p<\infty$ では下限と $p$ 乗が交換でき，
  \[
    \Wass_p(\mu,\nu)^p
    =\inf_{\pi\in\Couplings(\mu,\nu)}
    \int_{X\times X}d(x,y)^p\,\d\pi(x,y)
  \]
  とも書ける．
\end{remark}

\begin{lemma}{$\Wass_p$ の有限性}{wp-finite}
  $1\leq p\leq\infty$，$\mu,\nu\in\Pp{p}(X)$ とすると，
  $\Couplings(\mu,\nu)\neq\emptyset$ であり，
  任意の $\pi\in\Couplings(\mu,\nu)$ に対して
  $\norm{d}_{L^p(\pi)}<\infty$ である．
  特に $\Wass_p(\mu,\nu)<\infty$ である．
\end{lemma}

\begin{proof}
  直積測度 $\mu\otimes\nu$（定理~\ref{thm:meas-product}）は，
  主張~\ref{clm:meas-product-marginals}より
  第1周辺分布 $\mu$・第2周辺分布 $\nu$ をもつから，
  $\mu\otimes\nu\in\Couplings(\mu,\nu)$，
  特に $\Couplings(\mu,\nu)\neq\emptyset$ である．

  次に $\pi\in\Couplings(\mu,\nu)$ を任意に取る．
  各点で成り立つ三角不等式 $d(x,y)\leq d(x,x_0)+d(x_0,y)$ に，
  $L^p$ ノルムの単調性（主張~\ref{clm:meas-lp-monotone}）と
  Minkowski の不等式（定理~\ref{thm:meas-minkowski}）を適用し，
  さらに周辺積分公式（補題~\ref{lem:meas-marginal-integral}の $L^p$ 版）で
  各項を周辺測度のノルムに落とすと，$1\leq p\leq\infty$ で一括して
  \[
    \norm{d(x,y)}_{L^p(\pi)}
    \leq
    \norm{d(x,x_0)}_{L^p(\pi)}
    +\norm{d(x_0,y)}_{L^p(\pi)}
    =
    \norm{d(\cdot,x_0)}_{L^p(\mu)}
    +\norm{d(\cdot,x_0)}_{L^p(\nu)}
  \]
  を得る．あとは右辺の各項が有限であることを確かめればよい．

  $1\leq p<\infty$ のとき，
  $\norm{d(\cdot,x_0)}_{L^p(\mu)}
  =\bigl(\int_X d(x,x_0)^p\,\d\mu\bigr)^{1/p}<\infty$
  は $\mu\in\Pp{p}(X)$ の定義そのものである（$\nu$ も同様）．

  $p=\infty$ のとき，$R_\mu:=\sup_{x\in\supp\mu}d(x,x_0)$ とおくと
  $\mu\in\Pp{\infty}(X)$ より $R_\mu<\infty$ である．
  $X$ は可分だから台は全測度をもち（補題~\ref{lem:meas-support-full}）
  $\mu(\supp\mu)=1$，かつ $\supp\mu$ 上で $d(\cdot,x_0)\leq R_\mu$ だから，
  「$d(\cdot,x_0)\leq R_\mu$」は $\mu$-a.e. で成り立つ．
  よって $\norm{d(\cdot,x_0)}_{L^\infty(\mu)}\leq R_\mu<\infty$ である
  （$\nu$ も同様）．
\end{proof}

\begin{theorem}{最適 coupling の存在}{wp-optimal-coupling-existence}
  $1\leq p\leq\infty$，$\mu,\nu\in\Pp{p}(X)$ とする．
  このとき $\pi^*\in\Couplings(\mu,\nu)$ が存在して，
  $1\leq p<\infty$ なら
  \[
    \Wass_p(\mu,\nu)^p
    =
    \int_{X\times X}d(x,y)^p\,\d\pi^*(x,y),
  \]
  $p=\infty$ なら
  \[
    \Wass_\infty(\mu,\nu)
    =
    \norm{d}_{L^\infty(\pi^*)}
  \]
  をみたす．このような $\pi^*$ を $\mu$ から $\nu$ への
  \textbf{最適 coupling}または\textbf{最適輸送計画}という．
\end{theorem}

\begin{remark}{Villani の存在定理との関係}{wp-optimal-coupling-villani}
  $1\leq p<\infty$ の場合，定理~\ref{thm:wp-optimal-coupling-existence}は
  Villani (2009) Theorem 4.1 の特殊例である．
  実際，Villani の定理は2つの Polish 確率空間 $(S,\mu)$，
  $(T,\nu)$ と下半連続コスト
  $c\colon S\times T\to\R\cup\{+\infty\}$ を扱い，
  $c(x,y)\geq a(x)+b(y)$ となる可積分な下界があれば
  $\int c\,\d\pi$ を最小にする coupling が存在すると述べる．
  本資料では $S=T=X$，$c(x,y)=d(x,y)^p\geq0$ とすればよい．
  なお $p=\infty$ は積分型コストではなく本質的上限を最小化するので，
  下の証明で別に扱う．
\end{remark}

\begin{proof}
  直積 $X\times X$ は Polish 空間である
  （付録Aの直積距離の注意を参照）．まず
  $\Couplings(\mu,\nu)$ が tight であることを示す．
  $\varepsilon>0$ を任意に取る．Prokhorov の定理
  （定理~\ref{thm:meas-prokhorov}）より $\mu,\nu$ は tight だから，
  compact 集合 $K_1,K_2\subset X$ で
  \[
    \mu(K_1)\geq 1-\frac{\varepsilon}{2},
    \qquad
    \nu(K_2)\geq 1-\frac{\varepsilon}{2}
  \]
  なるものが取れる．このとき $K_1\times K_2$ は compact であり，
  任意の $\pi\in\Couplings(\mu,\nu)$ に対して
  \[
    \pi\bigl((K_1\times K_2)^c\bigr)
    \leq
    \pi(K_1^c\times X)+\pi(X\times K_2^c)
    =
    \mu(K_1^c)+\nu(K_2^c)
    \leq \varepsilon.
  \]
  よって $\Couplings(\mu,\nu)$ は tight である．

  次に，$\Couplings(\mu,\nu)$ は弱収束で閉じていることを確認する．
  $\pi_n\in\Couplings(\mu,\nu)$，$\pi_n\Rightarrow\pi$ とする．
  任意の $\varphi\in C_b(X)$ に対し，
  $\varphi\circ\mathrm{pr}_1\in C_b(X\times X)$ だから，
  弱収束と周辺積分公式（補題~\ref{lem:meas-marginal-integral}）より
  \[
    \int_X\varphi\,\d(\pi\circ\mathrm{pr}_1^{-1})
    =
    \int_{X\times X}\varphi(x)\,\d\pi(x,y)
    =
    \lim_{n\to\infty}\int_{X\times X}\varphi(x)\,\d\pi_n(x,y)
    =
    \int_X\varphi\,\d\mu.
  \]
  有界連続関数による測度の判定
  （定理~\ref{thm:meas-cb-determines}）より
  $\pi\circ\mathrm{pr}_1^{-1}=\mu$ である．同様に
  $\pi\circ\mathrm{pr}_2^{-1}=\nu$ である．したがって
  $\pi\in\Couplings(\mu,\nu)$ である．

  $1\leq p<\infty$ の場合を示す．
  \[
    I:=\inf_{\pi\in\Couplings(\mu,\nu)}
    \int_{X\times X}d(x,y)^p\,\d\pi(x,y)
  \]
  とおく．補題~\ref{lem:wp-finite}より $I<\infty$ である．
  下限の性質から，$\pi_n\in\Couplings(\mu,\nu)$ で
  \[
    \int d(x,y)^p\,\d\pi_n(x,y)\leq I+\frac1n
  \]
  をみたすものを取れる．先ほど示した tightness と Prokhorov の定理により，
  ある部分列 $(\pi_{n_k})$ と $\pi^*\in\Prob(X\times X)$ が存在して
  $\pi_{n_k}\Rightarrow\pi^*$ となる．弱閉性から
  $\pi^*\in\Couplings(\mu,\nu)$ である．
  関数 $(x,y)\mapsto d(x,y)^p$ は非負連続，したがって下半連続なので，
  Portmanteau 型の下半連続性（定理~\ref{thm:meas-portmanteau}）より
  \[
    \int d(x,y)^p\,\d\pi^*(x,y)
    \leq
    \liminf_{k\to\infty}
    \int d(x,y)^p\,\d\pi_{n_k}(x,y)
    \leq I.
  \]
  一方，$I$ は $\Couplings(\mu,\nu)$ 上の下限だから逆向きの不等式も成り立つ．
  よって $\pi^*$ は下限を達成する．注意~\ref{rem:wp-welldef}より，
  これは
  \[
    \Wass_p(\mu,\nu)^p
    =
    \int d(x,y)^p\,\d\pi^*(x,y)
  \]
  を意味する．

  最後に $p=\infty$ の場合を示す．
  $R:=\Wass_\infty(\mu,\nu)$ とおく．補題~\ref{lem:wp-finite}より
  $R<\infty$ である．下限の性質から，$\pi_n\in\Couplings(\mu,\nu)$ で
  \[
    \norm{d}_{L^\infty(\pi_n)}<R+\frac1n
  \]
  をみたすものを取れる．同じく Prokhorov の定理により，部分列を取って
  $\pi_{n_k}\Rightarrow\pi^*$ とでき，弱閉性から
  $\pi^*\in\Couplings(\mu,\nu)$ である．
  任意の $\delta>0$ に対し，$k$ が十分大きければ
  $R+1/n_k<R+\delta$ である．このとき
  主張~\ref{clm:meas-esssup-ae}より
  \[
    d(x,y)\leq \norm{d}_{L^\infty(\pi_{n_k})}<R+\delta
    \qquad \pi_{n_k}\text{-a.e.}
  \]
  であるから，閉集合
  \[
    F_\delta:=\{(x,y)\in X\times X\mid d(x,y)\leq R+\delta\}
  \]
  について $\pi_{n_k}(F_\delta)=1$ が十分大きい $k$ で成り立つ．
  Portmanteau 型の閉集合評価（定理~\ref{thm:meas-portmanteau}）より
  $\pi^*(F_\delta)=1$ である．$\delta=1/m$ として $m\to\infty$ とすれば，
  測度の上からの連続性により
  \[
    \pi^*(\{(x,y)\mid d(x,y)\leq R\})=1.
  \]
  したがって $\norm{d}_{L^\infty(\pi^*)}\leq R$ である．
  $R$ は下限なので逆向きの不等式も成り立ち，
  $\norm{d}_{L^\infty(\pi^*)}=R=\Wass_\infty(\mu,\nu)$ である．
\end{proof}

\section{Wasserstein 距離の距離性}
\label{sec:metric-property}

\begin{lemma}{接着補題（gluing lemma）}{wp-gluing}
  $\mu,\nu,\xi\in\Prob(X)$，$\pi\in\Couplings(\mu,\nu)$，
  $\sigma\in\Couplings(\nu,\xi)$ とする．
  このとき $X^3$ 上の確率測度 $\gamma$ で，
  $(x,y)$-周辺分布が $\pi$，$(y,z)$-周辺分布が $\sigma$
  であるものが存在する．
  ここで $(x,y)$-周辺分布とは，射影
  $\mathrm{pr}_{12}\colon(x,y,z)\mapsto(x,y)$
  （添字 $12$ は「第1・第2成分を残す」の意）による像測度
  $\gamma\circ\mathrm{pr}_{12}^{-1}$ のことをいう（$(y,z)$ も同様．
  $X^3$ には付録Aの直積距離を入れ，射影は連続ゆえ可測である）．
\end{lemma}

\begin{proof}
  方針：$\pi$ と $\sigma$ がともにもつ周辺分布 $\nu$ を軸に，
  $y$ で条件付けた2つの Markov 核を貼り合わせる．

  $X$ は Polish 空間だから，
  分解定理（定理~\ref{thm:meas-disintegration}）が使える．
  $\pi$ の第2周辺分布は $\nu$ だから，Markov 核 $K_\pi$ で
  \[
    \pi(A\times B)=\int_B K_\pi(y,A)\,\d\nu(y)
    \qquad(A,B\in\Borel(X))
  \]
  をみたすものが取れる．
  次に $s(y,z):=(z,y)$ とおくと（連続ゆえ可測），
  $(\sigma\circ s^{-1})(C\times B)
  =\sigma(s^{-1}(C\times B))=\sigma(B\times C)$ だから，
  $\sigma\circ s^{-1}$ の第2周辺分布は $\sigma$ の第1周辺分布，
  すなわち $\nu$ である．
  再び分解定理を $\sigma\circ s^{-1}$ に適用して，Markov 核 $K_\sigma$ で
  \[
    \sigma(B\times C)
    =(\sigma\circ s^{-1})(C\times B)
    =\int_B K_\sigma(y,C)\,\d\nu(y)
    \qquad(B,C\in\Borel(X))
  \]
  をみたすものが取れる．

  定理~\ref{thm:meas-kernel-composition}により，$X^3$ 上の確率測度
  \[
    \gamma(\d x,\d y,\d z)
    :=
    K_\pi(y,\d x)\,\nu(\d y)\,K_\sigma(y,\d z),
  \]
  すなわち
  $\gamma(A\times B\times C)=\int_B K_\pi(y,A)\,K_\sigma(y,C)\,\d\nu(y)$
  をみたす $\gamma$ が存在する．
  2つの周辺分布を検証する．
  ${\mathrm{pr}_{12}}^{-1}(A\times B)=A\times B\times X$ と
  $K_\sigma(y,X)=1$（Markov 核の値は確率測度）から，
  任意の $A,B\in\Borel(X)$ に対し
  \[
    (\gamma\circ\mathrm{pr}_{12}^{-1})(A\times B)
    =\gamma(A\times B\times X)
    =\int_B K_\pi(y,A)\,K_\sigma(y,X)\,\d\nu(y)
    =\int_B K_\pi(y,A)\,\d\nu(y)
    =\pi(A\times B).
  \]
  矩形集合の全体は $\Borel(X\times X)$ を生成する $\pi$-系だから
  （注意~\ref{rem:meas-borel-product}），
  一致定理（定理~\ref{thm:meas-uniqueness}）より
  $\gamma\circ\mathrm{pr}_{12}^{-1}=\pi$ である．
  同様に $K_\pi(y,X)=1$ より
  \[
    (\gamma\circ\mathrm{pr}_{23}^{-1})(B\times C)
    =\gamma(X\times B\times C)
    =\int_B K_\pi(y,X)\,K_\sigma(y,C)\,\d\nu(y)
    =\int_B K_\sigma(y,C)\,\d\nu(y)
    =\sigma(B\times C)
  \]
  だから $\gamma\circ\mathrm{pr}_{23}^{-1}=\sigma$ である．
\end{proof}

\begin{proposition}{Wasserstein 距離は距離である}{wp-metric}
  $1\leq p\leq\infty$ に対し，
  $\Wass_p$ は $\Pp{p}(X)$ 上の距離である．
\end{proposition}

\begin{proof}
  距離の公理（定義~\ref{def:meas-metric-space}）を順に確かめる．
  以下，統一記法
  $\Wass_p(\mu,\nu)=\inf_{\pi}\norm{d}_{L^p(\pi)}$
  （注意~\ref{rem:wp-welldef}）を $1\leq p\leq\infty$ で共通に用いる．

  \textbf{(0) 非負性と有限性．}
  任意の $\pi\in\Couplings(\mu,\nu)$ で $\norm{d}_{L^p(\pi)}\geq0$ だから
  $\Wass_p(\mu,\nu)\geq0$，
  また補題~\ref{lem:wp-finite}より
  $\Couplings(\mu,\nu)\neq\emptyset$ かつ $\Wass_p(\mu,\nu)<\infty$．
  よって $\Wass_p$ は $\Pp{p}(X)$ 上の $[0,\infty)$ に値をもつ関数である．

  \textbf{(1) 対称性．}
  $s(x,y):=(y,x)$ とおく（連続ゆえ可測）．
  $\pi\in\Couplings(\mu,\nu)$ に対し $\pi':=\pi\circ s^{-1}$ とおくと，
  \[
    \pi'(B\times X)=\pi(s^{-1}(B\times X))=\pi(X\times B)=\nu(B),
    \qquad
    \pi'(X\times A)=\pi(A\times X)=\mu(A)
  \]
  だから $\pi'\in\Couplings(\nu,\mu)$ である．
  コストは，系~\ref{cor:meas-pf-lp}と $d$ の対称性（$d\circ s=d$）より
  \[
    \norm{d}_{L^p(\pi')}
    =\norm{d\circ s}_{L^p(\pi)}
    =\norm{d}_{L^p(\pi)}.
  \]
  $s\circ s=\Id$ と合成則（主張~\ref{clm:meas-pf-compose}）より，
  対応 $\pi\mapsto \pi\circ s^{-1}$ は
  $\Couplings(\mu,\nu)$ から $\Couplings(\nu,\mu)$ への全単射だから，
  両辺の下限を取って $\Wass_p(\mu,\nu)=\Wass_p(\nu,\mu)$ である．

  \textbf{(2) 一致の公理．}
  まず $\Wass_p(\mu,\mu)=0$ を示す．
  対角写像 $\iota(x):=(x,x)$（連続ゆえ可測）による像測度
  $\pi^*:=\mu\circ\iota^{-1}$ は，$\iota^{-1}(A\times B)=A\cap B$ より
  \[
    \pi^*(A\times X)=\mu(A\cap X)=\mu(A),
    \qquad
    \pi^*(X\times B)=\mu(B)
  \]
  をみたすから $\pi^*\in\Couplings(\mu,\mu)$ である．
  $d\circ\iota\equiv0$ だから，系~\ref{cor:meas-pf-lp}より
  $1\leq p\leq\infty$ で一括して
  \[
    \norm{d}_{L^p(\pi^*)}=\norm{d\circ\iota}_{L^p(\mu)}=0,
  \]
  よって $\Wass_p(\mu,\mu)=0$ である．

  逆に $\Wass_p(\mu,\nu)=0$ とする．
  定理~\ref{thm:wp-optimal-coupling-existence}より，最適 coupling
  $\pi^*\in\Couplings(\mu,\nu)$ が取れる．
  $1\leq p<\infty$ なら
  \[
    \int_{X\times X}d(x,y)^p\,\d\pi^*(x,y)
    =\Wass_p(\mu,\nu)^p=0,
  \]
  であり，$p=\infty$ なら
  \[
    \norm{d}_{L^\infty(\pi^*)}
    =\Wass_\infty(\mu,\nu)=0
  \]
  である．いずれの場合も $d(x,y)=0$ が $\pi^*$-a.e. で成り立つ．
  距離の公理よりこれは $x=y$ と同値なので，対角集合
  \[
    \Delta:=\{(x,x)\mid x\in X\}
    =\{(x,y)\in X\times X\mid d(x,y)=0\}
  \]
  は $\pi^*$ の全質量をもつ．よって任意の $A\in\Borel(X)$ に対し，
  対角上では $A\times X$ と $X\times A$ が一致することから
  \[
    \mu(A)
    =\pi^*(A\times X)
    =\pi^*((A\times X)\cap\Delta)
    =\pi^*((X\times A)\cap\Delta)
    =\pi^*(X\times A)
    =\nu(A).
  \]
  したがって $\mu=\nu$ である．

  \textbf{(3) 三角不等式．}
  方針：最適 coupling $\pi,\sigma$ を gluing lemma で貼り合わせ，
  $(x,z)$-周辺分布を $\mu$ から $\xi$ への coupling として使う．

  $\mu,\nu,\xi\in\Pp{p}(X)$ を任意に取る．
  定理~\ref{thm:wp-optimal-coupling-existence}より，
  最適 coupling $\pi\in\Couplings(\mu,\nu)$ と
  $\sigma\in\Couplings(\nu,\xi)$ が取れる．
  補題~\ref{lem:wp-gluing}により，$X^3$ 上の確率測度 $\gamma$ で
  $\gamma\circ\mathrm{pr}_{12}^{-1}=\pi$，
  $\gamma\circ\mathrm{pr}_{23}^{-1}=\sigma$
  なるものが取れる．
  $\tau:=\gamma\circ\mathrm{pr}_{13}^{-1}$ とおくと，
  \[
    \tau(A\times X)=\gamma(A\times X\times X)=\pi(A\times X)=\mu(A),
    \qquad
    \tau(X\times C)=\gamma(X\times X\times C)=\sigma(X\times C)=\xi(C)
  \]
  だから $\tau\in\Couplings(\mu,\xi)$ であり，
  \[
    \Wass_p(\mu,\xi)
    \leq \norm{d}_{L^p(\tau)}.
  \]
  以下，$\gamma$ 上の $L^p$ ノルムでは，
  $d(x,z)$ 等を $(x,y,z)\mapsto d(x,z)$ の略記とする．
  系~\ref{cor:meas-pf-lp}（$T=\mathrm{pr}_{13}$）より
  \[
    \norm{d}_{L^p(\tau)}=\norm{d(x,z)}_{L^p(\gamma)}
  \]
  であり，距離の三角不等式 $d(x,z)\leq d(x,y)+d(y,z)$（各点）と
  $L^p$ ノルムの単調性（主張~\ref{clm:meas-lp-monotone}），
  続いて Minkowski の不等式（定理~\ref{thm:meas-minkowski}）から
  \[
    \norm{d(x,z)}_{L^p(\gamma)}
    \leq \norm{d(x,y)+d(y,z)}_{L^p(\gamma)}
    \leq \norm{d(x,y)}_{L^p(\gamma)}+\norm{d(y,z)}_{L^p(\gamma)}.
  \]
  再び系~\ref{cor:meas-pf-lp}（$T=\mathrm{pr}_{12}$ と $T=\mathrm{pr}_{23}$）より
  $\norm{d(x,y)}_{L^p(\gamma)}=\norm{d}_{L^p(\pi)}$，
  $\norm{d(y,z)}_{L^p(\gamma)}=\norm{d}_{L^p(\sigma)}$ だから，まとめて
  \[
    \Wass_p(\mu,\xi)
    \leq\norm{d}_{L^p(\pi)}+\norm{d}_{L^p(\sigma)}
    =\Wass_p(\mu,\nu)+\Wass_p(\nu,\xi)
  \]
  が従う．

  以上 (0)--(3) より，$\Wass_p$ は $\Pp{p}(X)$ 上の距離である．
\end{proof}

\begin{remark}{文献との対応}{wp-metric-references}
  命題~\ref{prop:wp-metric}は Givens--Shortt (1984) の Proposition 2 である．
  Villani (2009) Definition 6.1, 6.4 では，$1\leq p<\infty$ に対して
  まず $\Prob(X)$ 上に $\Wass_p$ を定義し，その後
  $\Pp{p}(X)=\{\mu\in\Prob(X)\mid \int d(x_0,x)^p\,\d\mu(x)<\infty\}$
  に制限すれば有限値の距離になる，という順序で述べている．
  三角不等式の証明は Villani (2009) Chapter 6 の ``Proof that
  $W_p$ satisfies the axioms of a distance'' と同じく，
  接着補題で2つの最適 coupling を $X^3$ 上に貼り合わせ，
  $d(x,z)\leq d(x,y)+d(y,z)$ と Minkowski の不等式を用いる．
  本資料でもこの形に合わせるため，
  Prokhorov の定理と Portmanteau 型の下半連続性を標準結果として認めたうえで，
  定理~\ref{thm:wp-optimal-coupling-existence}として
  Villani (2009) Theorem 4.1 の特殊例にあたる最適 coupling の存在を
  先に証明した．
  $\Pp{p}(X)$ 上での有限性は，Villani (2009) Definition 6.4 直後の
  \[
    d(x,y)^p\leq 2^{p-1}\bigl(d(x,x_0)^p+d(x_0,y)^p\bigr)
  \]
  による確認に対応する．
  接着補題（補題~\ref{lem:wp-gluing}）の構成は
  桑江ほか『最適輸送理論とリッチ曲率』
  （Encounter with Mathematics 第63回，2015）の補題 1.33，
  および Villani (2009) Chapter 1 の Gluing Lemma に対応する．
  $\Wass_p(\mu,\nu)=0\Rightarrow\mu=\nu$ については，
  Villani の同箇所と同じく，対角集合に集中する最適輸送計画を用いて示した．
  なお Villani の定義は $1\leq p<\infty$ であるが，本資料では
  Givens--Shortt (1984) に合わせて $p=\infty$ も同時に扱う．
\end{remark}
