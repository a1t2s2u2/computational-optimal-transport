\chapter{Wasserstein 距離 \texorpdfstring{$W_p$}{Wp}}
\label{ch:wasserstein-metrics}

\section{設定と定義}
\label{sec:wp-definition}

$(X,d)$ を Polish 空間
（完備かつ可分な距離空間：定義~\ref{def:meas-polish}）とする．
Borel $\sigma$-代数 $\Borel(X)$ 上の確率測度全体を
\[
  \Prob(X)
  :=
  \left\{
    \mu\colon\Borel(X)\to[0,1]
    \;\middle|\;
    \mu\text{ は測度},\;\mu(X)=1
  \right\}
\]
とおく．

\begin{definition}{有限 $p$ 次モーメントをもつ確率測度}{wp-p-space}
  任意に点 $x_0\in X$ を一つ固定する．$1\leq p<\infty$ に対し
  \[
    \Pp{p}(X)
    :=
    \left\{
      \mu\in\Prob(X)
      \;\middle|\;
      \int_X d(x,x_0)^p\,\d\mu(x)<\infty
    \right\}
  \]
  と定める．
\end{definition}

\begin{lemma}{$\Pp{p}(X)$ の $x_0$ 非依存性}{wp-p-indep}
  $1\leq p<\infty$ のとき，$\Pp{p}(X)$ は点 $x_0$ の選び方によらない．
\end{lemma}

\begin{proof}
  $x_0,x_0'\in X$ と $\mu\in\Prob(X)$ を任意に取り，
  $\int_X d(x,x_0)^p\,\d\mu(x)<\infty$ を仮定する．
  $a:=d(x,x_0)$，$b:=d(x_0,x_0')$ とおくと，三角不等式
  $d(x,x_0')\leq a+b$ と，$t\mapsto t^p$（$p\geq1$）が
  $[0,\infty)$ 上で凸であること（注意~\ref{rem:meas-real-facts}）から
  \[
    d(x,x_0')^p
    \leq (a+b)^p
    = 2^p\left(\frac{a}{2}+\frac{b}{2}\right)^p
    \leq 2^p\cdot\frac{a^p+b^p}{2}
    = 2^{p-1}\bigl(d(x,x_0)^p+d(x_0,x_0')^p\bigr).
  \]
  両辺を $\d\mu(x)$ で積分すると
  （右辺第2項は定数なので，$\mu(X)=1$ より積分値は $d(x_0,x_0')^p$）
  \[
    \int_X d(x,x_0')^p\,\d\mu(x)
    \leq
    2^{p-1}\left(\int_X d(x,x_0)^p\,\d\mu(x)+d(x_0,x_0')^p\right)
    <\infty.
  \]
  $x_0$ と $x_0'$ を入れ替えれば逆も成り立つ．
\end{proof}

\begin{definition}{Coupling 集合}{wp-coupling}
  距離空間 $S,T$ 上の確率測度 $\mu\in\Prob(S)$，
  $\nu\in\Prob(T)$ に対して，\textbf{Coupling 集合}を
  \[
    \Couplings(\mu,\nu)
    :=
    \left\{
      \pi\in\Prob(S\times T)
      \;\middle|\;
      \begin{aligned}
        \pi(A\times T)&=\mu(A) &&(\forall A\in\Borel(S)),\\
        \pi(S\times B)&=\nu(B) &&(\forall B\in\Borel(T))
      \end{aligned}
    \right\}
  \]
  と定める．
  $\pi\in\Couplings(\mu,\nu)$ は $\mu,\nu$ の
  \textbf{coupling}（結合分布）あるいは\textbf{輸送計画}とよばれる．
  この条件は「$\pi$ の第1周辺分布が $\mu$，第2周辺分布が $\nu$」，
  すなわち射影の像測度として
  $\pi\circ\mathrm{pr}_1^{-1}=\mu$，$\pi\circ\mathrm{pr}_2^{-1}=\nu$
  と言い換えられる（補題~\ref{lem:meas-marginal-integral}）．
\end{definition}

\begin{definition}{Wasserstein 距離}{wp-wasserstein}
  $1\leq p<\infty$ と $\mu,\nu\in\Pp{p}(X)$ に対し，
  下限（定義~\ref{def:meas-sup-inf}）を用いて
  \[
    \Wass_p(\mu,\nu)
    :=
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \left(
      \int_{X\times X}d(x,y)^p\,\d\pi(x,y)
    \right)^{1/p}
  \]
  と定める．
\end{definition}

\begin{remark}{記法と可測性}{wp-welldef}
  距離関数 $d\colon X\times X\to[0,\infty)$ は，三角不等式から従う評価
  \[
    \abs{d(x,y)-d(x',y')}\leq d(x,x')+d(y,y')
  \]
  により連続，したがって Borel 可測である
  （注意~\ref{rem:meas-measurable-ops}）．
  $d\geq0$ より
  $\norm{d}_{L^p(\pi)}=\bigl(\int d^p\,\d\pi\bigr)^{1/p}$
  （$L^p$ ノルムは定義~\ref{def:meas-lp-norm}）だから，
  \[
    \Wass_p(\mu,\nu)
    =\inf_{\pi\in\Couplings(\mu,\nu)}\norm{d}_{L^p(\pi)}
  \]
  と書ける．
  さらに $t\mapsto t^{1/p}$ が $[0,\infty)$ 上の単調な全単射であることから，
  $p<\infty$ では下限と $p$ 乗が交換でき，
  \[
    \Wass_p(\mu,\nu)^p
    =\inf_{\pi\in\Couplings(\mu,\nu)}
    \int_{X\times X}d(x,y)^p\,\d\pi(x,y)
  \]
  とも書ける．
\end{remark}

\begin{lemma}{$\Wass_p$ の有限性}{wp-finite}
  $1\leq p<\infty$，$\mu,\nu\in\Pp{p}(X)$ とすると，
  $\Couplings(\mu,\nu)\neq\emptyset$ であり，
  任意の $\pi\in\Couplings(\mu,\nu)$ に対して
  $\norm{d}_{L^p(\pi)}<\infty$ である．
  特に $\Wass_p(\mu,\nu)<\infty$ である．
\end{lemma}

\begin{proof}
  直積測度 $\mu\otimes\nu$（定理~\ref{thm:meas-product}）は，
  主張~\ref{clm:meas-product-marginals}より
  第1周辺分布 $\mu$・第2周辺分布 $\nu$ をもつから，
  $\mu\otimes\nu\in\Couplings(\mu,\nu)$，
  特に $\Couplings(\mu,\nu)\neq\emptyset$ である．

  次に $\pi\in\Couplings(\mu,\nu)$ を任意に取る．
  各点で成り立つ三角不等式 $d(x,y)\leq d(x,x_0)+d(x_0,y)$ に，
  $L^p$ ノルムの単調性（主張~\ref{clm:meas-lp-monotone}）と
  Minkowski の不等式（定理~\ref{thm:meas-minkowski}）を適用し，
  さらに周辺積分公式（補題~\ref{lem:meas-marginal-integral}の $L^p$ 版）で
  各項を周辺測度のノルムに落とすと
  \[
    \norm{d(x,y)}_{L^p(\pi)}
    \leq
    \norm{d(x,x_0)}_{L^p(\pi)}
    +\norm{d(x_0,y)}_{L^p(\pi)}
    =
    \norm{d(\cdot,x_0)}_{L^p(\mu)}
    +\norm{d(\cdot,x_0)}_{L^p(\nu)}
  \]
  を得る．ここで
  $\norm{d(\cdot,x_0)}_{L^p(\mu)}
  =\bigl(\int_X d(x,x_0)^p\,\d\mu\bigr)^{1/p}<\infty$
  は $\mu\in\Pp{p}(X)$ の定義そのものである（$\nu$ も同様）．
\end{proof}

\section{最適 coupling の存在}
\label{sec:wp-existence}

下限 $\Wass_p(\mu,\nu)$ を達成する coupling が存在することを，
Villani (2009) Theorem 4.1 の形で示す．

\begin{theorem}{最適 coupling の存在}{wp-optimal-coupling-existence}
  $(S,\mu)$，$(T,\nu)$ を2つの Polish 確率空間
  （Polish 空間とその上の Borel 確率測度の組をいう）とする．
  $a\colon S\to\R\cup\{-\infty\}$，
  $b\colon T\to\R\cup\{-\infty\}$ を上半連続関数とし，
  $a\in L^1(\mu)$，$b\in L^1(\nu)$，すなわち
  $\int_S\abs{a}\,\d\mu<\infty$，$\int_T\abs{b}\,\d\nu<\infty$ とする
  （$\abs{-\infty}:=+\infty$ と読む．
  特に $a$ は $\mu$-a.e. で，$b$ は $\nu$-a.e. で有限値である）．
  また，$c\colon S\times T\to\R\cup\{+\infty\}$ を下半連続関数とし，
  すべての $x\in S$，$y\in T$ で
  \[
    c(x,y)\geq a(x)+b(y)
  \]
  が成り立つとする．このとき，ある coupling
  $\pi^*\in\Couplings(\mu,\nu)$ が存在して
  \[
    \int_{S\times T}c\,\d\pi^*
    =
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \int_{S\times T}c\,\d\pi
  \]
  をみたす．このような $\pi^*$ をコスト $c$ に関する
  \textbf{最適 coupling}という．
\end{theorem}

\begin{remark}{下界条件の役割}{wp-cost-lower-bound}
  仮定 $c\geq a+b$ により，任意の $\pi\in\Couplings(\mu,\nu)$ で
  $c$ の負部分 $c^-:=\max\{-c,0\}$ は
  $c^-\leq(a+b)^-\leq\abs{a(x)}+\abs{b(y)}$ をみたし，
  周辺積分公式（補題~\ref{lem:meas-marginal-integral}）より
  $\pi$-可積分である
  （$c$ の可測性は半連続性による：注意~\ref{rem:meas-lsc-facts}）．
  したがって $\int c\,\d\pi$ は $\R\cup\{+\infty\}$ の元として定まる
  （Villani (2009) Remark 4.2）．
\end{remark}

\begin{proof}
  直積 $S\times T$ は Polish 空間であり
  （付録Aの直積距離の注意を参照），
  $\mu\otimes\nu\in\Couplings(\mu,\nu)$
  （主張~\ref{clm:meas-product-marginals}）だから
  $\Couplings(\mu,\nu)\neq\emptyset$ である．
  まず $\Couplings(\mu,\nu)$ が tight であることを示す．
  $\varepsilon>0$ を任意に取る．Prokhorov の定理
  （定理~\ref{thm:meas-prokhorov}）より $\mu,\nu$ は tight だから，
  compact 集合 $K_1\subset S$，$K_2\subset T$ で
  \[
    \mu(K_1)\geq 1-\frac{\varepsilon}{2},
    \qquad
    \nu(K_2)\geq 1-\frac{\varepsilon}{2}
  \]
  なるものが取れる．このとき $K_1\times K_2$ は compact であり，
  任意の $\pi\in\Couplings(\mu,\nu)$ に対して
  \[
    \pi\bigl((K_1\times K_2)^c\bigr)
    \leq
    \pi(K_1^c\times T)+\pi(S\times K_2^c)
    =
    \mu(K_1^c)+\nu(K_2^c)
    \leq \varepsilon.
  \]
  よって $\Couplings(\mu,\nu)$ は tight である．

  次に，$\Couplings(\mu,\nu)$ は弱収束で閉じていることを確認する．
  $\pi_n\in\Couplings(\mu,\nu)$，$\pi_n\Rightarrow\pi$ とする．
  任意の $\varphi\in C_b(S)$ に対し，
  $\varphi\circ\mathrm{pr}_1\in C_b(S\times T)$ だから，
  弱収束と周辺積分公式（補題~\ref{lem:meas-marginal-integral}）より
  \[
    \int_S\varphi\,\d(\pi\circ\mathrm{pr}_1^{-1})
    =
    \int_{S\times T}\varphi(x)\,\d\pi(x,y)
    =
    \lim_{n\to\infty}\int_{S\times T}\varphi(x)\,\d\pi_n(x,y)
    =
    \int_S\varphi\,\d\mu.
  \]
  有界連続関数による測度の判定
  （定理~\ref{thm:meas-cb-determines}）より
  $\pi\circ\mathrm{pr}_1^{-1}=\mu$ である．同様に
  $\pi\circ\mathrm{pr}_2^{-1}=\nu$ である．したがって
  $\pi\in\Couplings(\mu,\nu)$ である．

  \[
    I:=
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \int_{S\times T}c\,\d\pi
  \]
  とおく．任意の $\pi\in\Couplings(\mu,\nu)$ に対し，
  $c\geq a+b$ と周辺積分公式（補題~\ref{lem:meas-marginal-integral}）より
  \[
    \int_{S\times T}c\,\d\pi
    \geq\int_{S\times T}\bigl(a(x)+b(y)\bigr)\,\d\pi(x,y)
    =\int_S a\,\d\mu+\int_T b\,\d\nu
    >-\infty
  \]
  だから，$I>-\infty$ である．
  $I=+\infty$ のときはすべての coupling が最小値を達成する．
  以下 $I<+\infty$ とする．下限の性質から，
  $\pi_n\in\Couplings(\mu,\nu)$ で
  \[
    \int c\,\d\pi_n\leq I+\frac1n
  \]
  をみたすものを取れる．先ほど示した tightness と Prokhorov の定理により，
  ある部分列 $(\pi_{n_k})$ と $\pi^*\in\Prob(S\times T)$ が存在して
  $\pi_{n_k}\Rightarrow\pi^*$ となる．弱閉性から
  $\pi^*\in\Couplings(\mu,\nu)$ である．

  $h(x,y):=c(x,y)-a(x)-b(y)$ とおく．
  $-a(x)$，$-b(y)$ は下半連続関数と連続な射影の合成だから下半連続，
  したがって和 $h$ も下半連続である（注意~\ref{rem:meas-lsc-facts}．
  $a$ または $b$ が $-\infty$ の点では $h=+\infty$ と読む）．
  また $c\geq a+b$ より $h\geq0$ である．
  また任意の $\pi\in\Couplings(\mu,\nu)$ に対し，周辺積分公式より
  \[
    \int h\,\d\pi
    =
    \int c\,\d\pi
    -\int_S a\,\d\mu
    -\int_T b\,\d\nu
  \]
  である．Portmanteau 型の下半連続性
  （定理~\ref{thm:meas-portmanteau}）を $h$ に適用すると
  \[
    \int h\,\d\pi^*
    \leq
    \liminf_{k\to\infty}\int h\,\d\pi_{n_k}
  \]
  であり，両辺に
  $\int_Sa\,\d\mu+\int_Tb\,\d\nu$ を加えれば
  \[
    \int c\,\d\pi^*
    \leq
    \liminf_{k\to\infty}\int c\,\d\pi_{n_k}
    \leq I.
  \]
  一方，$I$ は下限なので逆向きの不等式も成り立つ．
  よって $\pi^*$ は最小値を達成する．
\end{proof}

\begin{corollary}{Wasserstein 距離の最適 coupling}{wp-wasserstein-optimal-coupling}
  $1\leq p<\infty$，$\mu,\nu\in\Pp{p}(X)$ とする．
  このとき $\pi^*\in\Couplings(\mu,\nu)$ が存在して
  \[
    \Wass_p(\mu,\nu)^p
    =
    \int_{X\times X}d(x,y)^p\,\d\pi^*(x,y)
  \]
  をみたす．
\end{corollary}

\begin{proof}
  定理~\ref{thm:wp-optimal-coupling-existence}を
  $S=T=X$，$c(x,y)=d(x,y)^p$，$a=b=0$ に適用すればよい
  （$c$ は連続ゆえ下半連続である）．
\end{proof}

\section{Wasserstein 距離の距離性}
\label{sec:metric-property}

\begin{lemma}{接着補題（gluing lemma）}{wp-gluing}
  $\mu,\nu,\xi\in\Prob(X)$，$\pi\in\Couplings(\mu,\nu)$，
  $\sigma\in\Couplings(\nu,\xi)$ とする．
  このとき $X^3$ 上の確率測度 $\gamma$ で，
  $(x,y)$-周辺分布が $\pi$，$(y,z)$-周辺分布が $\sigma$
  であるものが存在する．
  ここで $(x,y)$-周辺分布とは，射影
  $\mathrm{pr}_{12}\colon(x,y,z)\mapsto(x,y)$ による像測度
  $\gamma\circ\mathrm{pr}_{12}^{-1}$ のことをいう（$(y,z)$ も同様．
  $X^3$ には付録Aの直積距離を入れ，射影は連続ゆえ可測である）．
\end{lemma}

\begin{proof}
  方針：$\pi$ と $\sigma$ がともにもつ周辺分布 $\nu$ を軸に，
  $y$ で条件付けた2つの Markov 核を貼り合わせる．

  $X$ は Polish 空間だから，
  分解定理（定理~\ref{thm:meas-disintegration}）が使える．
  $\pi$ の第2周辺分布は $\nu$ だから，Markov 核 $K_\pi$ で
  \[
    \pi(A\times B)=\int_B K_\pi(y,A)\,\d\nu(y)
    \qquad(A,B\in\Borel(X))
  \]
  をみたすものが取れる．
  次に $s(y,z):=(z,y)$ とおくと（連続ゆえ可測），
  $(\sigma\circ s^{-1})(C\times B)
  =\sigma(s^{-1}(C\times B))=\sigma(B\times C)$ だから，
  $\sigma\circ s^{-1}$ の第2周辺分布は $\sigma$ の第1周辺分布，
  すなわち $\nu$ である．
  再び分解定理を $\sigma\circ s^{-1}$ に適用して，Markov 核 $K_\sigma$ で
  \[
    \sigma(B\times C)
    =(\sigma\circ s^{-1})(C\times B)
    =\int_B K_\sigma(y,C)\,\d\nu(y)
    \qquad(B,C\in\Borel(X))
  \]
  をみたすものが取れる．

  定理~\ref{thm:meas-kernel-composition}により，$X^3$ 上の確率測度
  \[
    \gamma(\d x,\d y,\d z)
    :=
    K_\pi(y,\d x)\,\nu(\d y)\,K_\sigma(y,\d z),
  \]
  すなわち
  $\gamma(A\times B\times C)=\int_B K_\pi(y,A)\,K_\sigma(y,C)\,\d\nu(y)$
  をみたす $\gamma$ が存在する．
  2つの周辺分布を検証する．
  ${\mathrm{pr}_{12}}^{-1}(A\times B)=A\times B\times X$ と
  $K_\sigma(y,X)=1$（Markov 核の値は確率測度）から，
  任意の $A,B\in\Borel(X)$ に対し
  \[
    (\gamma\circ\mathrm{pr}_{12}^{-1})(A\times B)
    =\gamma(A\times B\times X)
    =\int_B K_\pi(y,A)\,K_\sigma(y,X)\,\d\nu(y)
    =\int_B K_\pi(y,A)\,\d\nu(y)
    =\pi(A\times B).
  \]
  矩形集合の全体は $\Borel(X\times X)$ を生成する $\pi$-系だから
  （注意~\ref{rem:meas-borel-product}），
  一致定理（定理~\ref{thm:meas-uniqueness}）より
  $\gamma\circ\mathrm{pr}_{12}^{-1}=\pi$ である．
  同様に $K_\pi(y,X)=1$ より
  \[
    (\gamma\circ\mathrm{pr}_{23}^{-1})(B\times C)
    =\gamma(X\times B\times C)
    =\int_B K_\pi(y,X)\,K_\sigma(y,C)\,\d\nu(y)
    =\int_B K_\sigma(y,C)\,\d\nu(y)
    =\sigma(B\times C)
  \]
  だから $\gamma\circ\mathrm{pr}_{23}^{-1}=\sigma$ である．
\end{proof}

\begin{proposition}{Wasserstein 距離は距離である}{wp-metric}
  $1\leq p<\infty$ に対し，
  $\Wass_p$ は $\Pp{p}(X)$ 上の距離である．
\end{proposition}

\begin{proof}
  距離の公理（定義~\ref{def:meas-metric-space}）を順に確かめる．
  以下，記法
  $\Wass_p(\mu,\nu)=\inf_{\pi}\norm{d}_{L^p(\pi)}$
  （注意~\ref{rem:wp-welldef}）を用いる．

  \textbf{(0) 非負性と有限性．}
  任意の $\pi\in\Couplings(\mu,\nu)$ で $\norm{d}_{L^p(\pi)}\geq0$ だから
  $\Wass_p(\mu,\nu)\geq0$，
  また補題~\ref{lem:wp-finite}より
  $\Couplings(\mu,\nu)\neq\emptyset$ かつ $\Wass_p(\mu,\nu)<\infty$ である．

  \textbf{(1) 対称性．}
  $s(x,y):=(y,x)$ とおく（連続ゆえ可測）．
  $\pi\in\Couplings(\mu,\nu)$ に対し $\pi':=\pi\circ s^{-1}$ とおくと，
  \[
    \pi'(B\times X)=\pi(s^{-1}(B\times X))=\pi(X\times B)=\nu(B),
    \qquad
    \pi'(X\times A)=\pi(A\times X)=\mu(A)
  \]
  だから $\pi'\in\Couplings(\nu,\mu)$ である．
  コストは，系~\ref{cor:meas-pf-lp}と $d$ の対称性（$d\circ s=d$）より
  \[
    \norm{d}_{L^p(\pi')}
    =\norm{d\circ s}_{L^p(\pi)}
    =\norm{d}_{L^p(\pi)}.
  \]
  $s\circ s=\Id$ と合成則（主張~\ref{clm:meas-pf-compose}）より，
  対応 $\pi\mapsto \pi\circ s^{-1}$ は
  $\Couplings(\mu,\nu)$ から $\Couplings(\nu,\mu)$ への全単射だから，
  両辺の下限を取って $\Wass_p(\mu,\nu)=\Wass_p(\nu,\mu)$ である．

  \textbf{(2) 一致の公理．}
  まず $\Wass_p(\mu,\mu)=0$ を示す．
  対角写像 $\iota(x):=(x,x)$（連続ゆえ可測）による像測度
  $\pi^*:=\mu\circ\iota^{-1}$ は，$\iota^{-1}(A\times B)=A\cap B$ より
  \[
    \pi^*(A\times X)=\mu(A\cap X)=\mu(A),
    \qquad
    \pi^*(X\times B)=\mu(B)
  \]
  をみたすから $\pi^*\in\Couplings(\mu,\mu)$ である．
  $d\circ\iota\equiv0$ だから，系~\ref{cor:meas-pf-lp}より
  \[
    \norm{d}_{L^p(\pi^*)}=\norm{d\circ\iota}_{L^p(\mu)}=0,
  \]
  よって $\Wass_p(\mu,\mu)=0$ である．

  逆に $\Wass_p(\mu,\nu)=0$ とする．
  系~\ref{cor:wp-wasserstein-optimal-coupling}より，最適 coupling
  $\pi^*\in\Couplings(\mu,\nu)$ が取れる．
  \[
    \int_{X\times X}d(x,y)^p\,\d\pi^*(x,y)
    =\Wass_p(\mu,\nu)^p=0
  \]
  だから，非負可測関数の積分が $0$ ならばその関数が a.e. で $0$
  であることより，$d(x,y)=0$ が $\pi^*$-a.e. で成り立つ．
  距離の公理よりこれは $x=y$ と同値なので，対角集合
  \[
    \Delta:=\{(x,x)\mid x\in X\}
    =\{(x,y)\in X\times X\mid d(x,y)=0\}
  \]
  は $\pi^*$ の全質量をもつ．よって任意の $A\in\Borel(X)$ に対し，
  対角上では $A\times X$ と $X\times A$ が一致することから
  \[
    \mu(A)
    =\pi^*(A\times X)
    =\pi^*((A\times X)\cap\Delta)
    =\pi^*((X\times A)\cap\Delta)
    =\pi^*(X\times A)
    =\nu(A).
  \]
  したがって $\mu=\nu$ である．

  \textbf{(3) 三角不等式．}
  方針：最適 coupling $\pi,\sigma$ を gluing lemma で貼り合わせ，
  $(x,z)$-周辺分布を $\mu$ から $\xi$ への coupling として使う．

  $\mu,\nu,\xi\in\Pp{p}(X)$ を任意に取る．
  系~\ref{cor:wp-wasserstein-optimal-coupling}より，
  最適 coupling $\pi\in\Couplings(\mu,\nu)$ と
  $\sigma\in\Couplings(\nu,\xi)$ が取れる．
  補題~\ref{lem:wp-gluing}により，$X^3$ 上の確率測度 $\gamma$ で
  $\gamma\circ\mathrm{pr}_{12}^{-1}=\pi$，
  $\gamma\circ\mathrm{pr}_{23}^{-1}=\sigma$
  なるものが取れる．
  $\tau:=\gamma\circ\mathrm{pr}_{13}^{-1}$ とおくと，
  \[
    \tau(A\times X)=\gamma(A\times X\times X)=\pi(A\times X)=\mu(A),
    \qquad
    \tau(X\times C)=\gamma(X\times X\times C)=\sigma(X\times C)=\xi(C)
  \]
  だから $\tau\in\Couplings(\mu,\xi)$ であり，
  \[
    \Wass_p(\mu,\xi)
    \leq \norm{d}_{L^p(\tau)}.
  \]
  以下，$\gamma$ 上の $L^p$ ノルムでは，
  $d(x,z)$ 等を $(x,y,z)\mapsto d(x,z)$ の略記とする．
  系~\ref{cor:meas-pf-lp}（$T=\mathrm{pr}_{13}$）より
  \[
    \norm{d}_{L^p(\tau)}=\norm{d(x,z)}_{L^p(\gamma)}
  \]
  であり，距離の三角不等式 $d(x,z)\leq d(x,y)+d(y,z)$（各点）と
  $L^p$ ノルムの単調性（主張~\ref{clm:meas-lp-monotone}），
  続いて Minkowski の不等式（定理~\ref{thm:meas-minkowski}）から
  \[
    \norm{d(x,z)}_{L^p(\gamma)}
    \leq \norm{d(x,y)+d(y,z)}_{L^p(\gamma)}
    \leq \norm{d(x,y)}_{L^p(\gamma)}+\norm{d(y,z)}_{L^p(\gamma)}.
  \]
  再び系~\ref{cor:meas-pf-lp}（$T=\mathrm{pr}_{12}$ と $T=\mathrm{pr}_{23}$）より
  $\norm{d(x,y)}_{L^p(\gamma)}=\norm{d}_{L^p(\pi)}$，
  $\norm{d(y,z)}_{L^p(\gamma)}=\norm{d}_{L^p(\sigma)}$ だから，まとめて
  \[
    \Wass_p(\mu,\xi)
    \leq\norm{d}_{L^p(\pi)}+\norm{d}_{L^p(\sigma)}
    =\Wass_p(\mu,\nu)+\Wass_p(\nu,\xi)
  \]
  が従う．

  以上 (0)--(3) より，$\Wass_p$ は $\Pp{p}(X)$ 上の距離である．
\end{proof}

\begin{remark}{文献との対応}{wp-metric-references}
  命題~\ref{prop:wp-metric}は Givens--Shortt (1984) の
  Proposition 2（$1\leq p<\infty$ の場合）である．
  最適 coupling の存在定理（定理~\ref{thm:wp-optimal-coupling-existence}）は
  Villani (2009) Theorem 4.1，距離の公理の証明は
  同 Chapter 6 の ``Proof that $W_p$ satisfies the axioms of a distance''
  に従った．
  接着補題（補題~\ref{lem:wp-gluing}）は
  Villani (2009) Chapter 1 の Gluing Lemma，
  桑江ほか (2015) の補題 1.33 に対応する．
  なお Villani (2009) Definition 6.1, 6.4 は $1\leq p<\infty$ に対して
  まず $\Prob(X)$ 上に $\Wass_p$ を定義し，
  その後 $\Pp{p}(X)$ に制限して有限値の距離とするが，
  本資料は Givens--Shortt (1984) に合わせて
  はじめから $\Pp{p}(X)$ 上で定義する
  （同論文は $p=\infty$ の $\Wass_\infty$ も扱うが，
  本資料は Villani に合わせて $1\leq p<\infty$ のみ扱う）．
\end{remark}
