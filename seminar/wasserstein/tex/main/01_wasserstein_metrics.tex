\chapter{Wasserstein 距離 \texorpdfstring{$W_p$}{Wp}}
\label{ch:wasserstein-metrics}

\section{設定と定義}
\label{sec:wp-definition}

$(\mathcal{X},d)$ を Polish 空間（定義~\ref{def:meas-polish}）とする．
Borel $\sigma$-代数 $\Borel(\mathcal{X})$ 上の確率測度全体を
\[
  \Prob(\mathcal{X})
  :=
  \left\{
    \mu\colon\Borel(\mathcal{X})\to[0,1]
    \;\middle|\;
    \mu\text{ は測度},\;\mu(\mathcal{X})=1
  \right\}
\]
とおく．

\begin{definition}{有限 $p$ 次モーメントをもつ確率測度（Villani Def 6.4）}{wp-p-space}
  任意に点 $x_0\in \mathcal{X}$ を一つ固定する．$1\leq p<\infty$ に対し
  \[
    \Pp{p}(\mathcal{X})
    :=
    \left\{
      \mu\in\Prob(\mathcal{X})
      \;\middle|\;
      \int_\mathcal{X} d(x,x_0)^p\,\d\mu(x)<\infty
    \right\}
  \]
  と定める．積分 $\int_{\mathcal{X}}d(x,x_0)^p\,\d\mu(x)$ を
  $\mu$ の（$x_0$ を基点とする）\textbf{$p$ 次モーメント}という．
\end{definition}

\begin{lemma}{$\Pp{p}(\mathcal{X})$ の $x_0$ 非依存性（Villani Def 6.4）}{wp-p-indep}
  $1\leq p<\infty$ のとき，$\Pp{p}(\mathcal{X})$ は点 $x_0$ の選び方によらない．
  すなわち，任意の $x_0,x_0'\in \mathcal{X}$ と $\mu\in\Prob(\mathcal{X})$ に対して
  \[
    \int_\mathcal{X} d(x,x_0)^p\,\d\mu(x)<\infty
    \iff
    \int_\mathcal{X} d(x,x_0')^p\,\d\mu(x)<\infty
  \]
  が成り立つ．
\end{lemma}

\begin{proof}
  $x_0,x_0'\in \mathcal{X}$ と $\mu\in\Prob(\mathcal{X})$ を任意に取り，
  $\int_\mathcal{X} d(x,x_0)^p\,\d\mu(x)<\infty$ を仮定して，
  もう一方の積分が有限であることを示す．
  $a:=d(x,x_0)$，$b:=d(x_0,x_0')$ とおくと，三角不等式
  $d(x,x_0')\leq a+b$ と，$t\mapsto t^p$ が
  $[0,\infty)$ 上で凸であること（注意~\ref{rem:meas-real-facts}）から
  \[
    d(x,x_0')^p
    \leq (a+b)^p
    = 2^p\left(\frac{a}{2}+\frac{b}{2}\right)^p
    \leq 2^p\cdot\frac{a^p+b^p}{2}
    = 2^{p-1}\bigl(d(x,x_0)^p+d(x_0,x_0')^p\bigr).
  \]
  両辺を $\d\mu(x)$ で積分すると
  （右辺第2項は定数なので，$\mu(\mathcal{X})=1$ より積分値は $d(x_0,x_0')^p$）
  \[
    \int_\mathcal{X} d(x,x_0')^p\,\d\mu(x)
    \leq
    2^{p-1}\left(\int_\mathcal{X} d(x,x_0)^p\,\d\mu(x)+d(x_0,x_0')^p\right)
    <\infty
  \]
  を得る．$x_0$ と $x_0'$ を入れ替えれば逆向きの含意も同様に従う．
\end{proof}

\begin{definition}{Coupling（Villani Def 1.1）}{wp-coupling}
  距離空間 $\mathcal{X},\mathcal{Y}$ 上の確率測度
  $\mu\in\Prob(\mathcal{X})$，$\nu\in\Prob(\mathcal{Y})$ に対し
  \[
    \Couplings(\mu,\nu)
    :=
    \left\{
      \pi\in\Prob(\mathcal{X}\times\mathcal{Y})
      \;\middle|\;
      \begin{aligned}
        \pi(A\times\mathcal{Y})&=\mu(A) &&(\forall A\in\Borel(\mathcal{X})),\\
        \pi(\mathcal{X}\times B)&=\nu(B) &&(\forall B\in\Borel(\mathcal{Y}))
      \end{aligned}
    \right\}
  \]
  と定め，$\pi\in\Couplings(\mu,\nu)$ を $\mu$ と $\nu$ の
  \textbf{coupling} という．
\end{definition}

\begin{definition}{Wasserstein 距離（Villani Def 6.1）}{wp-wasserstein}
  $1\leq p<\infty$ と $\mu,\nu\in\Pp{p}(\mathcal{X})$ に対し
  \[
    \Wass_p(\mu,\nu)
    :=
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \left(
      \int_{\mathcal{X}\times \mathcal{X}}d(x,y)^p\,\d\pi(x,y)
    \right)^{1/p}
  \]
  と定める．
\end{definition}

\begin{remark}{記法と可測性}{wp-welldef}
  距離関数 $d\colon \mathcal{X}\times \mathcal{X}\to[0,\infty)$ は，三角不等式から従う評価
  \[
    \abs{d(x,y)-d(x',y')}\leq d(x,x')+d(y,y')
  \]
  により連続，したがって Borel 可測である
  （注意~\ref{rem:meas-measurable-ops}）．
  $d\geq0$ より
  $\norm{d}_{L^p(\pi)}=\bigl(\int d^p\,\d\pi\bigr)^{1/p}$
  （$L^p$ ノルムは定義~\ref{def:meas-lp-norm}）だから，
  \[
    \Wass_p(\mu,\nu)
    =\inf_{\pi\in\Couplings(\mu,\nu)}\norm{d}_{L^p(\pi)}
  \]
  と書ける．
\end{remark}

\begin{lemma}{$\Wass_p$ の有限性（Villani Def 6.4）}{wp-finite}
  $1\leq p<\infty$，$\mu,\nu\in\Pp{p}(\mathcal{X})$ とすると，
  $\Couplings(\mu,\nu)\neq\emptyset$ であり，
  任意の $\pi\in\Couplings(\mu,\nu)$ に対して
  $\norm{d}_{L^p(\pi)}<\infty$ である．
  特に $\Wass_p(\mu,\nu)<\infty$ である．
\end{lemma}

\begin{proof}
  $\mu\otimes\nu\in\Couplings(\mu,\nu)$
  （定理~\ref{thm:meas-product}，主張~\ref{clm:meas-product-marginals}）
  より $\Couplings(\mu,\nu)\neq\emptyset$ である．

  次に $\pi\in\Couplings(\mu,\nu)$ を任意に取る．
  各点で成り立つ三角不等式 $d(x,y)\leq d(x,x_0)+d(x_0,y)$ に，
  $L^p$ ノルムの単調性（主張~\ref{clm:meas-lp-monotone}）と
  Minkowski の不等式（定理~\ref{thm:meas-minkowski}）を適用し，
  さらに周辺積分公式（補題~\ref{lem:meas-marginal-integral}の $L^p$ 版）で
  各項を周辺測度のノルムに落とすと
  \[
    \norm{d(x,y)}_{L^p(\pi)}
    \leq
    \norm{d(x,x_0)}_{L^p(\pi)}
    +\norm{d(x_0,y)}_{L^p(\pi)}
    =
    \norm{d(\cdot,x_0)}_{L^p(\mu)}
    +\norm{d(\cdot,x_0)}_{L^p(\nu)}
  \]
  を得る．ここで
  $\norm{d(\cdot,x_0)}_{L^p(\mu)}
  =\bigl(\int_\mathcal{X} d(x,x_0)^p\,\d\mu\bigr)^{1/p}<\infty$
  は $\mu\in\Pp{p}(\mathcal{X})$ の定義そのものである（$\nu$ も同様）．
\end{proof}

\section{最適 coupling の存在}
\label{sec:wp-existence}

下限 $\Wass_p(\mu,\nu)$ を達成する coupling が存在することを，
Villani (2009) Theorem 4.1 の形で示す．
以下，Polish 空間とその上の Borel 確率測度の組を
\textbf{Polish 確率空間}とよぶ．
また拡張実数値関数 $a$ に対し，$a\in L^1(\mu)$ は
$\int\abs{a}\,\d\mu<\infty$ の意味で用いる
（$\abs{-\infty}:=+\infty$ と読む）．

\begin{lemma}{コスト汎関数の下半連続性（Villani Lem 4.3）}{wp-cost-lsc}
  $\mathcal{X},\mathcal{Y}$ を Polish 空間，
  $c\colon\mathcal{X}\times\mathcal{Y}\to\R\cup\{+\infty\}$
  を下半連続関数，
  $h\colon\mathcal{X}\times\mathcal{Y}\to\R\cup\{-\infty\}$ を
  $c\geq h$ をみたす上半連続関数とする．
  $\mathcal{X}\times\mathcal{Y}$ 上の確率測度の列 $(\pi_k)$ が
  $\pi\in\Prob(\mathcal{X}\times\mathcal{Y})$ に弱収束し，
  $h\in L^1(\pi_k)$（$\forall k$），$h\in L^1(\pi)$，かつ
  \[
    \int h\,\d\pi_k\to\int h\,\d\pi
    \qquad(k\to\infty)
  \]
  をみたすとする．このとき
  \[
    \int c\,\d\pi
    \leq
    \liminf_{k\to\infty}\int c\,\d\pi_k.
  \]
\end{lemma}

\begin{proof}
  $c\geq0$ の場合は，空間 $\mathcal{X}\times\mathcal{Y}$ 上で
  補題~\ref{lem:meas-weak-lsc}を $f=c$ に適用すればよい．

  一般の場合は，$c$ を $c-h$ に置き換えて $c\geq0$ の場合に帰着する．
  $-h$ は $(-\infty,+\infty]$ に値をもつ下半連続関数だから，
  $c-h$ は非負下半連続である（注意~\ref{rem:meas-lsc-facts}．
  $h=-\infty$ の点では $c-h=+\infty$ と読む）．
  $h$ は $\pi$-a.e. で有限（$h\in L^1(\pi)$）だから
  $c=(c-h)+h$ が $\pi$-a.e. で成り立つ．
  また $c^-\leq\abs{h}\in L^1(\pi)$ より
  $\int c\,\d\pi$ は $\R\cup\{+\infty\}$ の元として定まり，
  \[
    \int c\,\d\pi=\int(c-h)\,\d\pi+\int h\,\d\pi
  \]
  が成り立つ：$\int(c-h)\,\d\pi<\infty$ のときは
  $\abs{c}\leq(c-h)+\abs{h}$（$\pi$-a.e.）より $c\in L^1(\pi)$ で，
  積分の線形性から従う．
  $\int(c-h)\,\d\pi=+\infty$ のときは
  $c-h\leq c^++\abs{h}$（$\pi$-a.e.）より $\int c^+\,\d\pi=+\infty$，
  すなわち両辺とも $+\infty$ である．
  各 $\pi_k$ についても同様である．
  よって $c\geq0$ の場合の結果と
  $\int h\,\d\pi_k\to\int h\,\d\pi$ から
  \[
    \int c\,\d\pi
    =\int(c-h)\,\d\pi+\int h\,\d\pi
    \leq\liminf_{k\to\infty}\int(c-h)\,\d\pi_k
    +\lim_{k\to\infty}\int h\,\d\pi_k
    =\liminf_{k\to\infty}\int c\,\d\pi_k.  \qedhere
  \]
\end{proof}

\begin{lemma}{輸送計画の tightness（Villani Lem 4.4）}{wp-plans-tight}
  $\mathcal{X},\mathcal{Y}$ を Polish 空間とし，
  $\mathcal{P}\subset\Prob(\mathcal{X})$，
  $\mathcal{Q}\subset\Prob(\mathcal{Y})$ を
  tight な部分集合とする．このとき，
  周辺分布がそれぞれ $\mathcal{P}$，$\mathcal{Q}$ に属する
  coupling の全体
  \[
    \Couplings(\mathcal{P},\mathcal{Q})
    :=\bigcup_{\mu\in\mathcal{P},\,\nu\in\mathcal{Q}}
    \Couplings(\mu,\nu)
  \]
  は $\Prob(\mathcal{X}\times\mathcal{Y})$ で tight である．
\end{lemma}

\begin{proof}
  $\varepsilon>0$ を任意に取る．仮定より，
  $\mu\in\mathcal{P}$ の取り方によらない compact 集合
  $K_\varepsilon\subset\mathcal{X}$ で
  $\mu(\mathcal{X}\setminus K_\varepsilon)\leq\varepsilon$，
  同様に $\nu\in\mathcal{Q}$ の取り方によらない compact 集合
  $L_\varepsilon\subset\mathcal{Y}$ で
  $\nu(\mathcal{Y}\setminus L_\varepsilon)\leq\varepsilon$ が取れる．
  任意の $\mu\in\mathcal{P}$，$\nu\in\mathcal{Q}$，
  $\pi\in\Couplings(\mu,\nu)$ に対し
  \[
    \pi\bigl((K_\varepsilon\times L_\varepsilon)^c\bigr)
    \leq
    \pi\bigl((\mathcal{X}\setminus K_\varepsilon)\times\mathcal{Y}\bigr)
    +\pi\bigl(\mathcal{X}\times(\mathcal{Y}\setminus L_\varepsilon)\bigr)
    =\mu(\mathcal{X}\setminus K_\varepsilon)
    +\nu(\mathcal{Y}\setminus L_\varepsilon)
    \leq 2\varepsilon.
  \]
  この評価は coupling の取り方によらず，
  $K_\varepsilon\times L_\varepsilon$ は
  $\mathcal{X}\times\mathcal{Y}$ の compact 集合
  （定義~\ref{def:meas-continuous-compact}）であり，
  $\varepsilon>0$ は任意だから，
  $\Couplings(\mathcal{P},\mathcal{Q})$ は tight である．
\end{proof}

\begin{theorem}{最適 coupling の存在（Villani Thm 4.1）}{wp-optimal-coupling-existence}
  $(\mathcal{X},\mu)$，$(\mathcal{Y},\nu)$ を2つの Polish 確率空間とする．
  $a\colon \mathcal{X}\to\R\cup\{-\infty\}$，
  $b\colon \mathcal{Y}\to\R\cup\{-\infty\}$ を
  $a\in L^1(\mu)$，$b\in L^1(\nu)$ をみたす上半連続関数とし，
  $c\colon \mathcal{X}\times\mathcal{Y}\to\R\cup\{+\infty\}$ を，
  すべての $x\in\mathcal{X}$，$y\in\mathcal{Y}$ で
  \[
    c(x,y)\geq a(x)+b(y)
  \]
  をみたす下半連続関数とする．このとき，ある coupling
  $\pi^*\in\Couplings(\mu,\nu)$ が存在して
  \[
    \int_{\mathcal{X}\times\mathcal{Y}}c\,\d\pi^*
    =
    \inf_{\pi\in\Couplings(\mu,\nu)}
    \int_{\mathcal{X}\times\mathcal{Y}}c\,\d\pi
  \]
  をみたす．このような $\pi^*$ をコスト $c$ に関する
  \textbf{最適 coupling}という．
\end{theorem}

\begin{proof}
  1点集合 $\{\mu\}$，$\{\nu\}$ は tight である
  （Prokhorov の定理~\ref{thm:meas-prokhorov}．
  1点集合は明らかに precompact）．
  よって補題~\ref{lem:wp-plans-tight}より，$\Couplings(\mu,\nu)$ は
  $\Prob(\mathcal{X}\times\mathcal{Y})$ で tight である
  （直積 $\mathcal{X}\times\mathcal{Y}$ は Polish 空間：
  付録Aの直積距離の注意を参照）．
  また $\mu\otimes\nu\in\Couplings(\mu,\nu)$
  （主張~\ref{clm:meas-product-marginals}）だから
  $\Couplings(\mu,\nu)\neq\emptyset$ である．

  下限の性質から，$\pi_k\in\Couplings(\mu,\nu)$ で
  \[
    \int c\,\d\pi_k
    \to
    I:=\inf_{\pi\in\Couplings(\mu,\nu)}
    \int_{\mathcal{X}\times\mathcal{Y}}c\,\d\pi
    \qquad(k\to\infty)
  \]
  となる列を取れる．$\Couplings(\mu,\nu)$ は tight だから，
  Prokhorov の定理より弱位相で precompact であり，
  部分列に移って，ある
  $\pi^*\in\Prob(\mathcal{X}\times\mathcal{Y})$ に
  $\pi_k\Rightarrow\pi^*$ としてよい．

  周辺分布の条件は弱収束で極限に移る：
  任意の $\varphi\in C_b(\mathcal{X})$ に対し，
  $\varphi\circ\mathrm{pr}_1\in C_b(\mathcal{X}\times\mathcal{Y})$ だから，
  弱収束と周辺積分公式（補題~\ref{lem:meas-marginal-integral}）より
  \[
    \int_{\mathcal{X}}\varphi\,\d(\pi^*\circ\mathrm{pr}_1^{-1})
    =
    \int_{\mathcal{X}\times\mathcal{Y}}\varphi(x)\,\d\pi^*(x,y)
    =
    \lim_{k\to\infty}\int_{\mathcal{X}\times\mathcal{Y}}\varphi(x)\,\d\pi_k(x,y)
    =
    \int_{\mathcal{X}}\varphi\,\d\mu
  \]
  であり，有界連続関数による測度の判定
  （定理~\ref{thm:meas-cb-determines}）より
  $\pi^*\circ\mathrm{pr}_1^{-1}=\mu$，同様に
  $\pi^*\circ\mathrm{pr}_2^{-1}=\nu$ である．
  したがって $\pi^*\in\Couplings(\mu,\nu)$ である．

  $h(x,y):=a(x)+b(y)$ とおくと，$h$ は上半連続で
  （$-a(x)$，$-b(y)$ は下半連続関数と連続な射影の合成，
  その和も下半連続：注意~\ref{rem:meas-lsc-facts}），
  仮定より $c\geq h$ である．
  また $\abs{h(x,y)}\leq\abs{a(x)}+\abs{b(y)}$ と周辺積分公式より，
  任意の $\sigma\in\Couplings(\mu,\nu)$ に対し
  \[
    \int\abs{h}\,\d\sigma
    \leq\int_{\mathcal{X}}\abs{a}\,\d\mu
    +\int_{\mathcal{Y}}\abs{b}\,\d\nu<\infty,
    \qquad
    \int h\,\d\sigma
    =\int_{\mathcal{X}}a\,\d\mu+\int_{\mathcal{Y}}b\,\d\nu
  \]
  である（拡張実数値の $a$ には，周辺積分公式の非負版を
  正部分・負部分 $a^\pm$ に適用して差を取る．$b$ も同様）．特に $h\in L^1(\pi_k)$（$\forall k$），$h\in L^1(\pi^*)$ で，
  $\int h\,\d\pi_k$ は $k$ によらない定数として
  $\int h\,\d\pi^*$ に収束する．
  よって補題~\ref{lem:wp-cost-lsc}より
  \[
    \int c\,\d\pi^*
    \leq
    \liminf_{k\to\infty}\int c\,\d\pi_k
    =I
  \]
  であり，$\pi^*\in\Couplings(\mu,\nu)$ より
  逆向きの不等式も成り立つ．
  よって $\pi^*$ は最小値を達成する．
\end{proof}

\begin{remark}{下界条件の役割（Villani Rem 4.2）}{wp-cost-lower-bound}
  下界条件 $c\geq a+b$（$a\in L^1(\mu)$，$b\in L^1(\nu)$）により，
  任意の $\pi\in\Couplings(\mu,\nu)$ で $c$ の負部分は
  周辺積分公式によって $\pi$-可積分となる．
  したがって下限の対象となる積分 $\int c\,\d\pi$ は，
  $\Couplings(\mu,\nu)$ 全体で $\R\cup\{+\infty\}$ の元として定まる．
\end{remark}

\begin{corollary}{Wasserstein 距離の最適 coupling}{wp-wasserstein-optimal-coupling}
  $1\leq p<\infty$，$\mu,\nu\in\Pp{p}(\mathcal{X})$ とする．
  このとき $\pi^*\in\Couplings(\mu,\nu)$ が存在して
  \[
    \Wass_p(\mu,\nu)^p
    =
    \int_{\mathcal{X}\times \mathcal{X}}d(x,y)^p\,\d\pi^*(x,y)
  \]
  をみたす．
\end{corollary}

\begin{proof}
  定理~\ref{thm:wp-optimal-coupling-existence}を
  $\mathcal{Y}=\mathcal{X}$，$c(x,y)=d(x,y)^p$，$a=b=0$ に適用すると
  （$c$ は連続ゆえ下半連続である），
  $\int d^p\,\d\pi$ を最小化する $\pi^*\in\Couplings(\mu,\nu)$ が取れる．
  $t\mapsto t^{1/p}$ は $[0,\infty)$ 上の単調な全単射だから
  下限と $p$ 乗は交換でき，
  \[
    \Wass_p(\mu,\nu)^p
    =\inf_{\pi\in\Couplings(\mu,\nu)}
    \int_{\mathcal{X}\times \mathcal{X}}d(x,y)^p\,\d\pi(x,y)
    =\int_{\mathcal{X}\times \mathcal{X}}d(x,y)^p\,\d\pi^*(x,y)
  \]
  である．
\end{proof}

\section{Wasserstein 距離の距離性}
\label{sec:metric-property}

\begin{lemma}{接着補題（gluing lemma）}{wp-gluing}
  $\mu,\nu,\xi\in\Prob(\mathcal{X})$，$\pi\in\Couplings(\mu,\nu)$，
  $\sigma\in\Couplings(\nu,\xi)$ とする．
  このとき $\mathcal{X}^3$ 上の確率測度 $\gamma$ で，
  $(x,y)$-周辺分布が $\pi$，$(y,z)$-周辺分布が $\sigma$
  であるものが存在する．
  ここで $(x,y)$-周辺分布とは，射影
  $\mathrm{pr}_{12}\colon(x,y,z)\mapsto(x,y)$ による像測度
  $\gamma\circ\mathrm{pr}_{12}^{-1}$ のことをいう（$(y,z)$ も同様．
  $\mathcal{X}^3$ には付録Aの直積距離を入れ，射影は連続ゆえ可測である）．
\end{lemma}

\begin{proof}
  方針：$\pi$ と $\sigma$ がともにもつ周辺分布 $\nu$ を軸に，
  $y$ で条件付けた2つの Markov 核を貼り合わせる．

  $\mathcal{X}$ は Polish 空間だから，
  分解定理（定理~\ref{thm:meas-disintegration}）が使える．
  $\pi$ の第2周辺分布は $\nu$ だから，Markov 核 $K_\pi$ で
  \[
    \pi(A\times B)=\int_B K_\pi(y,A)\,\d\nu(y)
    \qquad(A,B\in\Borel(\mathcal{X}))
  \]
  をみたすものが取れる．
  次に $s(y,z):=(z,y)$ とおくと（連続ゆえ可測），
  $(\sigma\circ s^{-1})(C\times B)
  =\sigma(s^{-1}(C\times B))=\sigma(B\times C)$ だから，
  $\sigma\circ s^{-1}$ の第2周辺分布は $\sigma$ の第1周辺分布，
  すなわち $\nu$ である．
  再び分解定理を $\sigma\circ s^{-1}$ に適用して，Markov 核 $K_\sigma$ で
  \[
    \sigma(B\times C)
    =(\sigma\circ s^{-1})(C\times B)
    =\int_B K_\sigma(y,C)\,\d\nu(y)
    \qquad(B,C\in\Borel(\mathcal{X}))
  \]
  をみたすものが取れる．

  定理~\ref{thm:meas-kernel-composition}により，$\mathcal{X}^3$ 上の確率測度
  \[
    \gamma(\d x,\d y,\d z)
    :=
    K_\pi(y,\d x)\,\nu(\d y)\,K_\sigma(y,\d z),
  \]
  すなわち
  $\gamma(A\times B\times C)=\int_B K_\pi(y,A)\,K_\sigma(y,C)\,\d\nu(y)$
  をみたす $\gamma$ が存在する．
  2つの周辺分布を検証する．
  ${\mathrm{pr}_{12}}^{-1}(A\times B)=A\times B\times \mathcal{X}$ と
  $K_\sigma(y,\mathcal{X})=1$（Markov 核の値は確率測度）から，
  任意の $A,B\in\Borel(\mathcal{X})$ に対し
  \[
    (\gamma\circ\mathrm{pr}_{12}^{-1})(A\times B)
    =\gamma(A\times B\times \mathcal{X})
    =\int_B K_\pi(y,A)\,K_\sigma(y,\mathcal{X})\,\d\nu(y)
    =\int_B K_\pi(y,A)\,\d\nu(y)
    =\pi(A\times B).
  \]
  矩形集合の全体は $\Borel(\mathcal{X}\times \mathcal{X})$ を生成する $\pi$-系だから
  （注意~\ref{rem:meas-borel-product}），
  一致定理（定理~\ref{thm:meas-uniqueness}）より
  $\gamma\circ\mathrm{pr}_{12}^{-1}=\pi$ である．
  同様に $K_\pi(y,\mathcal{X})=1$ より
  \[
    (\gamma\circ\mathrm{pr}_{23}^{-1})(B\times C)
    =\gamma(\mathcal{X}\times B\times C)
    =\int_B K_\pi(y,\mathcal{X})\,K_\sigma(y,C)\,\d\nu(y)
    =\int_B K_\sigma(y,C)\,\d\nu(y)
    =\sigma(B\times C)
  \]
  だから $\gamma\circ\mathrm{pr}_{23}^{-1}=\sigma$ である．
\end{proof}

\begin{proposition}{Wasserstein 距離は距離である（Villani Def 6.4）}{wp-metric}
  $1\leq p<\infty$ に対し，
  $\Wass_p$ は $\Pp{p}(\mathcal{X})$ 上の距離である．
\end{proposition}

\begin{proof}
  距離の公理（定義~\ref{def:meas-metric-space}）を順に確かめる．
  以下，記法
  $\Wass_p(\mu,\nu)=\inf_{\pi}\norm{d}_{L^p(\pi)}$
  （注意~\ref{rem:wp-welldef}）を用いる．

  \textbf{(0) 非負性と有限性．}
  任意の $\pi\in\Couplings(\mu,\nu)$ で $\norm{d}_{L^p(\pi)}\geq0$ だから
  $\Wass_p(\mu,\nu)\geq0$，
  また補題~\ref{lem:wp-finite}より
  $\Couplings(\mu,\nu)\neq\emptyset$ かつ $\Wass_p(\mu,\nu)<\infty$ である．

  \textbf{(1) 対称性．}
  $s(x,y):=(y,x)$ とおく（連続ゆえ可測）．
  $\pi\in\Couplings(\mu,\nu)$ に対し $\pi':=\pi\circ s^{-1}$ とおくと，
  \[
    \pi'(B\times \mathcal{X})=\pi(s^{-1}(B\times \mathcal{X}))=\pi(\mathcal{X}\times B)=\nu(B),
    \qquad
    \pi'(\mathcal{X}\times A)=\pi(A\times \mathcal{X})=\mu(A)
  \]
  だから $\pi'\in\Couplings(\nu,\mu)$ である．
  コストは，系~\ref{cor:meas-pf-lp}と $d$ の対称性（$d\circ s=d$）より
  \[
    \norm{d}_{L^p(\pi')}
    =\norm{d\circ s}_{L^p(\pi)}
    =\norm{d}_{L^p(\pi)}.
  \]
  $s\circ s=\Id$ と合成則（主張~\ref{clm:meas-pf-compose}）より，
  対応 $\pi\mapsto \pi\circ s^{-1}$ は
  $\Couplings(\mu,\nu)$ から $\Couplings(\nu,\mu)$ への全単射だから，
  両辺の下限を取って $\Wass_p(\mu,\nu)=\Wass_p(\nu,\mu)$ である．

  \textbf{(2) 一致の公理．}
  まず $\Wass_p(\mu,\mu)=0$ を示す．
  対角写像 $\iota(x):=(x,x)$（連続ゆえ可測）による像測度
  $\pi^*:=\mu\circ\iota^{-1}$ は，$\iota^{-1}(A\times B)=A\cap B$ より
  \[
    \pi^*(A\times \mathcal{X})=\mu(A\cap \mathcal{X})=\mu(A),
    \qquad
    \pi^*(\mathcal{X}\times B)=\mu(B)
  \]
  をみたすから $\pi^*\in\Couplings(\mu,\mu)$ である．
  $d\circ\iota\equiv0$ だから，系~\ref{cor:meas-pf-lp}より
  \[
    \norm{d}_{L^p(\pi^*)}=\norm{d\circ\iota}_{L^p(\mu)}=0,
  \]
  よって $\Wass_p(\mu,\mu)=0$ である．

  逆に $\Wass_p(\mu,\nu)=0$ とする．
  系~\ref{cor:wp-wasserstein-optimal-coupling}より，最適 coupling
  $\pi^*\in\Couplings(\mu,\nu)$ が取れる．
  \[
    \int_{\mathcal{X}\times \mathcal{X}}d(x,y)^p\,\d\pi^*(x,y)
    =\Wass_p(\mu,\nu)^p=0
  \]
  だから，非負可測関数の積分が $0$ ならばその関数が a.e. で $0$
  であることより，$d(x,y)=0$ が $\pi^*$-a.e. で成り立つ．
  距離の公理よりこれは $x=y$ と同値なので，対角集合
  \[
    \Delta:=\{(x,x)\mid x\in \mathcal{X}\}
    =\{(x,y)\in \mathcal{X}\times \mathcal{X}\mid d(x,y)=0\}
  \]
  は $\pi^*$ の全質量をもつ．よって任意の $A\in\Borel(\mathcal{X})$ に対し，
  対角上では $A\times \mathcal{X}$ と $\mathcal{X}\times A$ が一致することから
  \[
    \mu(A)
    =\pi^*(A\times \mathcal{X})
    =\pi^*((A\times \mathcal{X})\cap\Delta)
    =\pi^*((\mathcal{X}\times A)\cap\Delta)
    =\pi^*(\mathcal{X}\times A)
    =\nu(A).
  \]
  したがって $\mu=\nu$ である．

  \textbf{(3) 三角不等式．}
  $\mu,\nu,\xi\in\Pp{p}(\mathcal{X})$ を任意に取る．
  系~\ref{cor:wp-wasserstein-optimal-coupling}より，
  最適 coupling $\pi\in\Couplings(\mu,\nu)$ と
  $\sigma\in\Couplings(\nu,\xi)$ が取れる．
  補題~\ref{lem:wp-gluing}により，$\mathcal{X}^3$ 上の確率測度 $\gamma$ で
  $\gamma\circ\mathrm{pr}_{12}^{-1}=\pi$，
  $\gamma\circ\mathrm{pr}_{23}^{-1}=\sigma$
  なるものが取れる．
  $\tau:=\gamma\circ\mathrm{pr}_{13}^{-1}$ とおくと，
  \[
    \tau(A\times \mathcal{X})=\gamma(A\times \mathcal{X}\times \mathcal{X})=\pi(A\times \mathcal{X})=\mu(A),
    \qquad
    \tau(\mathcal{X}\times C)=\gamma(\mathcal{X}\times \mathcal{X}\times C)=\sigma(\mathcal{X}\times C)=\xi(C)
  \]
  だから $\tau\in\Couplings(\mu,\xi)$ であり，
  \[
    \Wass_p(\mu,\xi)
    \leq \norm{d}_{L^p(\tau)}.
  \]
  以下，$\gamma$ 上の $L^p$ ノルムでは，
  $d(x,z)$ 等を $(x,y,z)\mapsto d(x,z)$ の略記とする．
  系~\ref{cor:meas-pf-lp}（$T=\mathrm{pr}_{13}$）より
  \[
    \norm{d}_{L^p(\tau)}=\norm{d(x,z)}_{L^p(\gamma)}
  \]
  であり，距離の三角不等式 $d(x,z)\leq d(x,y)+d(y,z)$（各点）と
  $L^p$ ノルムの単調性（主張~\ref{clm:meas-lp-monotone}），
  続いて Minkowski の不等式（定理~\ref{thm:meas-minkowski}）から
  \[
    \norm{d(x,z)}_{L^p(\gamma)}
    \leq \norm{d(x,y)+d(y,z)}_{L^p(\gamma)}
    \leq \norm{d(x,y)}_{L^p(\gamma)}+\norm{d(y,z)}_{L^p(\gamma)}.
  \]
  再び系~\ref{cor:meas-pf-lp}（$T=\mathrm{pr}_{12}$ と $T=\mathrm{pr}_{23}$）より
  $\norm{d(x,y)}_{L^p(\gamma)}=\norm{d}_{L^p(\pi)}$，
  $\norm{d(y,z)}_{L^p(\gamma)}=\norm{d}_{L^p(\sigma)}$ だから，まとめて
  \[
    \Wass_p(\mu,\xi)
    \leq\norm{d}_{L^p(\pi)}+\norm{d}_{L^p(\sigma)}
    =\Wass_p(\mu,\nu)+\Wass_p(\nu,\xi)
  \]
  が従う．
\end{proof}

\begin{definition}{Wasserstein 空間（Villani Def 6.4）}{wp-wasserstein-space}
  距離 $\Wass_p$ を備えた距離空間
  $(\Pp{p}(\mathcal{X}),\Wass_p)$ を
  $p$ 次の \textbf{Wasserstein 空間}という．
\end{definition}
